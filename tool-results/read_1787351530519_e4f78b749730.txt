   337→// ───────────────────────────────────────────────────────────────────────────────
   338→// Duration Shortcuts — quick date/time selection with prices
   339→// ───────────────────────────────────────────────────────────────────────────────
   340→function DurationShortcuts({
   341→  startDate,
   342→  endDate,
   343→  startTime,
   344→  endTime,
   345→  onStartDateChange,
   346→  onEndDateChange,
   347→  onStartTimeChange,
   348→  onEndTimeChange,
   349→  T,
   350→  specs,
   351→}: {
   352→  startDate: string;
   353→  endDate: string;
   354→  startTime: string;
   355→  endTime: string;
   356→  onStartDateChange: (v: string) => void;
   357→  onEndDateChange: (v: string) => void;
   358→  onStartTimeChange: (v: string) => void;
   359→  onEndTimeChange: (v: string) => void;
   360→  T: CrewTokens;
   361→  specs?: Record<string, unknown>;
   362→}) {
   363→  const fmt = (n: number) => n.toLocaleString("ru-RU");
   364→  const today = todayISO();
   365→
   366→  const addHours = (timeStr: string, hours: number): string => {
   367→    const [h, m] = timeStr.split(":").map(Number);
   368→    const totalMinutes = (h * 60 + m) + hours * 60;
   369→    const newH = Math.floor(totalMinutes / 60) % 24;
   370→    const newM = totalMinutes % 60;
   371→    return `${String(newH).padStart(2, "0")}:${String(newM).padStart(2, "0")}`;
   372→  };
   373→
   374→  /** Check if an ISO date string falls on a weekend */
   375→  const isWeekendDay = (dateStr: string): boolean => {
   376→    if (!dateStr) return false;
   377→    const d = new Date(dateStr + "T12:00:00");
   378→    const day = d.getDay();
   379→    return day === 0 || day === 6;
   380→  };
   381→
   382→  const getDailyPrice = (): number => {
   383→    // Weekend override: rent_weekend when it's a Saturday/Sunday
   384→    if (isWeekendDay(startDate)) {
   385→      return Number(specs?.rent_weekend) || Number(specs?.dailyPrice) || Number(specs?.rent_weekday) || 0;
   386→    }
   387→    return Number(specs?.dailyPrice) || Number(specs?.rent_weekday) || 0;
   388→  };
   389→
   390→  const getPrice = (hours: number): number => {
   391→    if (!specs) return 0;
   392→    if (hours <= 1) return Number(specs?.price_per_hour) || 0;
   393→    if (hours === 2) return Number(specs?.price_per_2h) || Math.round((Number(specs?.price_per_hour) || 0) + ((Number(specs?.price_per_3h) || 0) - (Number(specs?.price_per_hour) || 0)) / 2) || (Number(specs?.price_per_hour) || 0) * 2;
   394→    if (hours <= 3) return Number(specs?.price_per_3h) || (Number(specs?.price_per_hour) || 0) * 3;
   395→    if (hours <= 6) return Number(specs?.price_per_6h) || (Number(specs?.price_per_hour) || 0) * 6;
   396→    if (hours <= 12) return Number(specs?.price_per_12h) || (Number(specs?.price_per_hour) || 0) * 12;
   397→    const days = Math.ceil(hours / 24);
   398→    const dailyPrice = getDailyPrice();
   399→    if (days === 1) return dailyPrice;
   400→    if (days <= 4) return (Number(specs?.rent_2_4d) || dailyPrice) * days;
   401→    if (days <= 10) return (Number(specs?.rent_5_10d) || dailyPrice) * days;
   402→    return (Number(specs?.rent_11_30d) || dailyPrice) * days;
   403→  };
   404→
   405→  const hourOptions = [
   406→    { label: "1 час", hours: 1 },
   407→    { label: "2 часа", hours: 2 },
   408→    { label: "3 часа", hours: 3 },
   409→    { label: "6 часов", hours: 6 },
   410→    { label: "12 часов", hours: 12 },
   411→  ];
   412→
   413→  const dayOptions = [
   414→    { label: "1 день", days: 1 },
   415→    { label: "3 дня", days: 3 },
   416→    { label: "7 дней", days: 7 },
   417→    { label: "2 недели", days: 14 },
   418→    { label: "1 месяц", days: 30 },
   419→  ];
   420→
   421→  const handleHourClick = (hours: number) => {
   422→    const effectiveStart = startDate || today;
   423→    onStartDateChange(effectiveStart);
   424→    onEndDateChange(effectiveStart);
   425→    onStartTimeChange(startTime);
   426→    onEndTimeChange(addHours(startTime, hours));
   427→  };
   428→
   429→  const handleDayClick = (days: number) => {
   430→    const effectiveStart = startDate || today;
   431→    onStartDateChange(effectiveStart);
   432→    onEndDateChange(addDaysISO(effectiveStart, days));
   433→    onStartTimeChange(startTime);
   434→    onEndTimeChange(startTime);
   435→  };
   436→
   437→  const buttonStyle = (isActive: boolean): React.CSSProperties => ({
   438→    borderColor: isActive ? T.accent : T.border,
   439→    color: isActive ? T.accent : T.text,
   440→    backgroundColor: isActive ? T.accentSoft : "transparent",
   441→  });
   442→
   443→  // FIX: Mutual exclusion between hour and day balloons. Previously
   444→  // `isDayActive(1)` returned `true` for a same-day rental because
   445→  // `diffDaysISO` is inclusive (it counts both the start and end day,
   446→  // so a same-day rental resolves to 1). That made both "3 часа" and
   447→  // "1 день" light up at the same time when the user clicked either
   448→  // one — the user reported this as "3 hours and 12 hours both get
   449→  // selected" because the day balloon was visually next to the
   450→  // hour balloons and they read it as a second hour option.
   451→  //
   452→  // The fix is twofold:
   453→  //   1. `isHourActive` — unchanged, but now ONLY fires on same-day
   454→  //      rentals. The minute math already guarantees mutual exclusion
   455→  //      between 3/6/12 because their minute counts (180/360/720) are
   456→  //      pairwise different.
   457→  //   2. `isDayActive` — now refuses to fire on same-day rentals
   458→  //      (those are the hour balloons' territory) and uses a NON-
   459→  //      inclusive day count so a 1-day rental (endDate = startDate
   460→  //      + 1) resolves to `d === 1`, not `d === 2`.
   461→  const isHourActive = (h: number): boolean => {
   462→    if (!endDate || !startDate || endDate !== startDate) return false;
   463→    const [sh, sm] = startTime.split(":").map(Number);
   464→    const [eh, em] = endTime.split(":").map(Number);
   465→    const minutes = (eh * 60 + em) - (sh * 60 + sm);
   466→    return Number.isFinite(minutes) && minutes === h * 60;
   467→  };
   468→
   469→  const isDayActive = (d: number): boolean => {
   470→    if (!startDate || !endDate) return false;
   471→    // Same-day rentals belong to the hour balloons, not the day ones.
   472→    if (endDate === startDate) return false;
   473→    try {
   474→      const { diffDaysISO } = require("@/app/franchize/lib/date-utils");
   475→      // Non-inclusive day count: a 1-day rental is endDate = startDate
   476→      // + 1, so `diffDaysISO - 1 === 1`. Without the -1 we'd hit
   477→      // `diffDaysISO === 1` which a 0-day (same-day) rental also
   478→      // satisfies — that's the bug that was lighting up "1 день"
   479→      // whenever the user picked any hour option.
   480→      return diffDaysISO(startDate, endDate) - 1 === d;
   481→    } catch {
   482→      return false;
   483→    }
   484→  };
   485→
   486→  const hasDates = !!startDate;
   487→
   488→  // Compute total price for the currently selected period
   489→  const totalPrice = (() => {
   490→    if (!hasDates || !endDate) return null;
   491→    try {
   492→      const { durationDaysFromDateTime } = require("@/app/franchize/lib/date-utils");
   493→      const days = durationDaysFromDateTime(startDate, startTime || "10:00", endDate, endTime || "10:00");
   494→      const totalHours = Math.round(days * 24);
   495→      const p = getPrice(totalHours);
   496→      if (p > 0) return p;
   497→    } catch {}
   498→    return null;
   499→  })();
   500→
   501→  return (
   502→    <div
   503→      className="rounded-2xl border p-3"
   504→      style={{ borderColor: T.borderSoft, backgroundColor: T.bgElevated }}
   505→    >
   506→      <div className="mb-2 flex items-center justify-between gap-2">
   507→        <p
   508→          className="flex items-center gap-1.5 text-xs font-medium uppercase tracking-[0.12em]"
   509→          style={{ color: T.textMuted }}
   510→        >
   511→          <Clock className="h-3.5 w-3.5" />
   512→          Быстрый выбор
   513→        </p>
   514→        {hasDates && totalPrice !== null && (
   515→          <p className="text-sm font-bold" style={{ color: T.accent }}>

   629→  dailyPrice,
   630→  startDate,
   631→  endDate,
   632→  borderColor,

   740→/** Get the effective price for an additional item based on rental duration. */
   741→function getAdditionalItemPrice(item: typeof ADDITIONAL_ITEMS[number], rentalHours?: number): number {
   742→  if (item.key === "helmet" && rentalHours !== undefined) {
   743→    return rentalHours < 24 ? (item.hourlyPrice ?? 500) : item.price;
   744→  }
   745→  return item.price;
   746→}

   748→/** Calculate total extras cost from selection.
   749→ *  @param rentalHours - if provided, helmet price is dynamic (500₽ for <24h, 1000₽ for ≥24h) */
   750→export function calcExtrasTotal(sel: AdditionalItemsSelection, rentalHours?: number): number {
   751→  return ADDITIONAL_ITEMS.reduce((sum, item) => {
   752→    const val = sel[item.key];
   753→    const unitPrice = getAdditionalItemPrice(item, rentalHours);
   754→    if (item.type === "count") return sum + (typeof val === "number" ? val : 0) * unitPrice;
   755→    return sum + (val === true ? unitPrice : 0);
   756→  }, 0);
   757→}

   759→/** Build a human-readable summary of selected extras. */
   760→export function extrasSummary(sel: AdditionalItemsSelection): string {
   761→  const parts: string[] = [];
   762→  for (const item of ADDITIONAL_ITEMS) {
   763→    const val = sel[item.key];
   764→    if (item.type === "count" && typeof val === "number" && val > 0) {
   765→      parts.push(`${item.label} ×${val}`);
   766→    } else if (val === true) {
   767→      parts.push(item.label);
   768→    }
   769→  }
   770→  return parts.join(", ");
   771→}

   902→function PriceCard({
   903→  specs,
   904→  startDate,
   905→  endDate,
   906→  startTime,
   907→  endTime,
   908→  helmetCount,
   909→  extrasSelection,
   910→  isExpanded,
   911→  onToggleExpand,
   912→  borderColor,
   913→}: {
   914→  specs: Record<string, unknown>;
   915→  startDate: string;
   916→  endDate: string;
   917→  startTime: string;
   918→  endTime: string;
   919→  helmetCount: number;
   920→  extrasSelection?: AdditionalItemsSelection;
   921→  isExpanded: boolean;
   922→  onToggleExpand: () => void;
   923→  borderColor: string;
   924→}) {
   925→  if (!startDate || !endDate) return null;
   926→
   927→  // Dynamic import of calculatePrice (client-side only)
   928→  const { calculatePrice, getHelmetPrice } = require("@/lib/rental-pricing-calculator");
   929→  const result = calculatePrice(
   930→    specs as any,
   931→    startDate,
   932→    endDate,
   933→    startTime || "10:00",
   934→    endTime || "10:00",
   935→    helmetCount
   936→  );
   937→
   938→  // Calculate rental hours for dynamic helmet pricing
   939→  const rentalHours = (() => {
   940→    try {
   941→      const start = new Date(`${startDate}T${startTime || "10:00"}`);
   942→      const end = new Date(`${endDate}T${endTime || "10:00"}`);
   943→      return Math.max(1, (end.getTime() - start.getTime()) / (1000 * 60 * 60));
   944→    } catch {
   945→      return 24; // default to daily pricing if dates are invalid
   946→    }
   947→  })();
   948→  const helmetUnitPrice = getHelmetPrice(rentalHours);
   949→
   950→  // Add non-helmet extras to the total
   951→  const extrasTotal = extrasSelection ? calcExtrasTotal(extrasSelection, rentalHours) : 0;
   952→  // Subtract helmet cost from extrasTotal (already counted in result.helmetRub)
   953→  const helmetFromExtras = extrasSelection ? (typeof extrasSelection.helmet === "number" ? (extrasSelection.helmet as number) * helmetUnitPrice : 0) : 0;
   954→  const nonHelmetExtras = extrasTotal - helmetFromExtras;
   955→  const grandTotal = result.totalRub + nonHelmetExtras;
   956→
   957→  const fmt = (n: number) => n.toLocaleString("ru-RU");
   958→
   959→  return (
   960→    <div
   961→      className="rounded-2xl border border-[var(--item-border)] bg-[var(--item-border)]/15 p-3"
   962→      style={{ borderColor }}
   963→    >
   964→      <div className="flex items-center justify-between gap-3">
   965→        <div className="flex items-center gap-2">
   966→          <CircleDollarSign className="h-5 w-5 text-[var(--item-accent)]" />
   967→          <span className="text-lg font-bold text-[var(--item-accent)]">
   968→            {fmt(grandTotal)} ₽
   969→          </span>
   970→        </div>
   971→        {result.savingsPercent > 0 && (
   972→          <span className="text-xs text-[var(--item-muted-text)]">
   973→            Экономия {result.savingsPercent}% vs посуточно
   974→          </span>
   975→        )}
   976→      </div>
   977→
   978→      {!isExpanded && (
   979→        <button
   980→          type="button"
   981→          onClick={onToggleExpand}
   982→          className="mt-2 text-xs text-[var(--item-accent)] transition hover:opacity-90"
   983→        >
   984→          Детали стоимости
   985→        </button>
   986→      )}
   987→
   988→      {isExpanded && (
   989→        <div className="mt-3 space-y-2 text-xs">
   990→          <p className="flex items-center gap-1.5 font-medium text-[var(--item-accent)]">
   991→            <BarChart3 className="h-4 w-4" /> Детали стоимости
   992→          </p>
   993→          <div className="space-y-1">
   994→            <p>• Период: {result.breakdown.period}</p>
   995→            <p>• Тариф: {result.breakdown.ratePerPeriod}</p>
   996→            <p>• Аренда: {fmt(result.basePriceRub)} ₽</p>
   997→            {result.helmetRub > 0 && <p>• Шлем ×{helmetCount}: {fmt(result.helmetRub)} ₽ ({fmt(helmetUnitPrice)} ₽/шт{rentalHours < 24 ? ", почасово" : ""})</p>}
   998→            {nonHelmetExtras > 0 && extrasSelection && (
   999→              <>
  1000→                {ADDITIONAL_ITEMS.filter((i) => i.key !== "helmet").map((item) => {
  1001→                  const val = extrasSelection[item.key];
  1002→                  if (val !== true && val !== 1) return null;
  1003→                  return (
  1004→                    <p key={item.key}>
  1005→                      • {item.label}: {item.price > 0 ? `${fmt(item.price)} ₽` : "бесплатно"}
  1006→                    </p>
  1007→                  );
  1008→                })}
  1009→              </>
  1010→            )}
  1011→            <p>• Залог: {fmt(result.depositRub)} ₽</p>
  1012→          </div>
  1013→          <div className="border-t border-[var(--item-border)] pt-2">
  1014→            <p className="font-semibold">
  1015→              Итого: {fmt(grandTotal)} ₽ (аренда

  1258→// Rounding Warning — shows when hours are messy (4-5h → 6h, 7-11h → 12h)
  1259→// ───────────────────────────────────────────────────────────────────────────────
  1260→function RoundingWarning({
  1261→  rounded,
  1262→  displayHours,
  1263→  startTime,
  1264→  onFixClick,
  1265→}: {
  1266→  rounded: boolean;
  1267→  displayHours?: number;
  1268→  startTime: string;
  1269→  onFixClick: () => void;
  1270→}) {
  1271→  if (!rounded || !displayHours) return null;
  1272→
  1273→  // Calculate rounded end time
  1274→  const [h, m] = startTime.split(":").map(Number);
  1275→  const totalMinutes = h * 60 + m + displayHours * 60;
  1276→  const newH = Math.floor(totalMinutes / 60) % 24;
  1277→  const newM = totalMinutes % 60;
  1278→  const roundedEndTime = `${String(newH).padStart(2, "0")}:${String(newM).padStart(2, "0")}`;
  1279→
  1280→  return (
  1281→    <div className="rounded-lg border border-amber-500/40 bg-amber-500/10 px-3 py-2 text-xs">
  1282→      <div className="flex items-center justify-between gap-3">
  1283→        <span className="inline-flex items-center gap-1 text-amber-300">
  1284→          <AlertTriangle className="h-3.5 w-3.5" aria-hidden /> Округляем до {displayHours} ч
  1285→        </span>
  1286→        <button
  1287→          type="button"
  1288→          onClick={onFixClick}
  1289→          className="shrink-0 rounded-full border border-amber-500/40 px-2 py-1 text-amber-300 hover:bg-amber-500/20 transition"
  1290→        >
  1291→          До {roundedEndTime}
  1292→        </button>
  1293→      </div>
  1294→    </div>
  1295→  );
  1296→}
