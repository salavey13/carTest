   122→async function resolveBikeById(bikeId: string): Promise<any> {
   123→  const { data: exactMatch } = await supabaseAdmin
   124→    .from("cars")
   125→    .select("id, make, model, specs, type")
   126→    .eq("id", bikeId)
   127→    .in("type", ["bike", "ebike"])
   128→    .maybeSingle();
   129→  if (exactMatch) return exactMatch;
   130→
   131→  const { data: candidates } = await supabaseAdmin
   132→    .from("cars")
   133→    .select("id, make, model, specs, type")
   134→    .in("type", ["bike", "ebike"])
   135→    .limit(100);
   136→  if (!candidates?.length) return null;
   137→
   138→  const norm = (v = "") => String(v).toLowerCase().replace(/[^a-zа-я0-9]+/gi, " ").trim();
   139→  const qn = norm(bikeId);
   140→  if (!qn) return null;
   141→
   142→  let best: any = null, bestScore = 0;
   143→  for (const bike of candidates) {
   144→    const hay = [bike.id, bike.make, bike.model, bike.specs?.vin, bike.specs?.frame].map(norm).join(" ");
   145→    if (hay.includes(qn)) { best = bike; bestScore = 1000; break; }
   146→    const parts = qn.split(" ");
   147→    let score = 0;
   148→    for (const p of parts) if (p && hay.includes(p)) score += 20 + p.length;
   149→    if (score > bestScore) { bestScore = score; best = bike; }
   150→  }
   151→  return bestScore > 0 ? best : null;
   152→}

   223→function parsePassport(text: string): { series: string; number: string; issueDate: string; issuedBy: string } | null {
   224→  const parts = text.trim().split(/\s+/);
   225→  if (parts.length < 3) return null;
   226→
   227→  let series = "", number = "", dateIdx = -1;
   228→  for (let i = 0; i < parts.length; i++) {
   229→    const cleaned = parts[i].replace(/\D/g, '');
   230→    if (!series && cleaned.length === 4) series = cleaned;
   231→    else if (!number && cleaned.length === 6) number = cleaned;
   232→    else if (dateIdx === -1 && /^\d{1,2}\.\d{1,2}(\.\d{2,4})?$/.test(parts[i])) dateIdx = i;
   233→  }
   234→
   235→  if (!series || !number || dateIdx === -1) return null;
   236→
   237→  let dateStr = parts[dateIdx];
   238→  const dateParts = dateStr.split('.');
   239→  if (dateParts.length === 2) return null;
   240→  if (dateParts[2].length === 2) {
   241→    const y = parseInt(dateParts[2]);
   242→    dateParts[2] = y > 50 ? `19${y}` : `20${y}`;
   243→    dateStr = dateParts.join('.');
   244→  }
   245→
   246→  const issuedBy = parts.slice(dateIdx + 1).join(' ') || "не указано";
   247→  return { series, number, issueDate: dateStr, issuedBy };
   248→}

   250→/**
   251→ * Parse Russian VU (driver's license) series + number + dates.
   252→ * Ported from doc-manual.ts parseLicense() with testdrive's 4-strategy series parsing.
   253→ *
   254→ * Format: серия номер [дата_выдачи [дата_окончания]]
   255→ *   "9976 123456 15.03"          → series=9976, num=123456, issue=15.03.2026, expiry=15.03.2036
   256→ *   "99 76 123456 15.03.2020 15.03.2030"  → series=9976, num=123456, issue=15.03.2020, expiry=15.03.2030
   257→ *   "99 76 123456"               → no dates → still returns series+number with null dates
   258→ *   "99 76123456 15.03"          → Strategy 4: 2+8 split → series=9976, num=123456, issue=15.03.2026
   259→ *
   260→ * If only one date: it's treated as issue date, expiry = issue + 10 years.
   261→ * Year defaults to CURRENT_YEAR if omitted.
   262→ */
   263→function parseLicense(text: string): { series: string; number: string; issueDate: string | null; expiryDate: string | null } | null {
   264→  const parts = text.trim().split(/\s+/).filter(Boolean);
   265→  if (parts.length < 2) return null;
   266→
   267→  // Separate digit sequences from date-like tokens
   268→  const digits: string[] = [];
   269→  const dates: string[] = [];
   270→
   271→  for (const p of parts) {
   272→    if (/^\d{1,2}\.\d{1,2}(\.\d{2,4})?$/.test(p)) {
   273→      // It's a date-like token
   274→      let d = p;
   275→      if (!d.includes('.')) continue;
   276→      const dparts = d.split('.');
   277→      if (dparts.length === 2) d += `.${CURRENT_YEAR}`;
   278→      else if (dparts[2].length === 2) {
   279→        const y = parseInt(dparts[2]);
   280→        dparts[2] = y > 50 ? `19${y}` : `20${y}`;
   281→        d = dparts.join('.');
   282→      }
   283→      dates.push(d);
   284→    } else {
   285→      const cleaned = p.replace(/\D/g, '');
   286→      if (cleaned.length > 0) digits.push(cleaned);
   287→    }
   288→  }
   289→
   290→  if (digits.length < 2) return null;
   291→
   292→  let series = "", number = "";
   293→
   294→  // Strategy 1: find a 4-digit (series) followed by a 6-digit (number)
   295→  for (let i = 0; i < digits.length; i++) {
   296→    if (digits[i].length === 4) {
   297→      series = digits[i];
   298→      for (let j = i + 1; j < digits.length; j++) {
   299→        if (digits[j].length === 6) { number = digits[j]; break; }
   300→      }
   301→      if (number) break;
   302→    }
   303→  }
   304→
   305→  // Strategy 2: two consecutive 2-digit tokens = series (e.g. "99 76"), then 6-digit = number
   306→  if (!series || !number) {
   307→    for (let i = 0; i < digits.length - 1; i++) {
   308→      if (digits[i].length === 2 && digits[i + 1].length === 2) {
   309→        series = digits[i] + digits[i + 1];
   310→        for (let j = i + 2; j < digits.length; j++) {
   311→          if (digits[j].length === 6) { number = digits[j]; break; }
   312→        }
   313→        if (number) break;
   314→      }
   315→    }
   316→  }
   317→
   318→  // Strategy 3: find any 4+ digit for series, any 6-digit for number
   319→  if (!series || !number) {
   320→    for (const d of digits) {
   321→      if (!series && d.length >= 4) series = d.slice(0, 4);
   322→      else if (!number && d.length === 6) number = d;
   323→    }
   324→  }
   325→
   326→  // Strategy 4: "99 76123456" -> 2-char + 8-char (where 8 = 2+6 concatenated)
   327→  if (!series || !number) {
   328→    for (let i = 0; i < digits.length - 1; i++) {
   329→      if (digits[i].length === 2 && digits[i + 1].length === 8) {
   330→        series = digits[i] + digits[i + 1].slice(0, 2);
   331→        number = digits[i + 1].slice(2);
   332→        if (number.length === 6) break;
   333→        series = ""; number = "";
   334→      }
   335→    }
   336→  }
   337→
   338→  if (!series || !number) return null;
   339→
   340→  // ── Date handling (ported from doc-manual.ts) ──
   341→  let issueDate: string | null = null;
   342→  let expiryDate: string | null = null;
   343→
   344→  if (dates.length === 1) {
   345→    // Single date → issue date, expiry = +10 years
   346→    issueDate = dates[0];
   347→    const [dd, mm, yyyy] = issueDate.split('.');
   348→    expiryDate = `${dd}.${mm}.${parseInt(yyyy) + 10}`;
   349→  } else if (dates.length >= 2) {
   350→    // Two dates → issue and expiry
   351→    issueDate = dates[0];
   352→    expiryDate = dates[1];
   353→    // If both same year → auto extend (e.g., "15.03.2026 15.03.2026" → +10y)
   354→    if (issueDate === expiryDate && issueDate.endsWith(`.${CURRENT_YEAR}`)) {
   355→      const [dd, mm, yyyy] = issueDate.split('.');
   356→      expiryDate = `${dd}.${mm}.${parseInt(yyyy) + 10}`;
   357→    }
   358→  }
   359→  // If no dates → issueDate/expiryDate stay null (template {{#if}} hides them)
   360→
   361→  return { series, number, issueDate, expiryDate };
   362→}

   364→function parseDate(text: string, requireYear = true): string | null {
   365→  const match = text.trim().match(/^(\d{1,2})\.(\d{1,2})(\.(\d{2,4}))?$/);
   366→  if (!match) return null;
   367→
   368→  let [, day, month, , year] = match;
   369→  day = day.padStart(2, '0');
   370→  month = month.padStart(2, '0');
   371→
   372→  if (!year) {
   373→    if (requireYear) return null;
   374→    year = String(CURRENT_YEAR);
   375→  } else if (year.length === 2) {
   376→    const y = parseInt(year);
   377→    year = y > 50 ? `19${y}` : `20${y}`;
   378→  }
   379→
   380→  return `${day}.${month}.${year}`;
   381→}

   383→function capitalizeFullName(text: string): string {
   384→  return text
   385→    .trim()
   386→    .replace(/\s+/g, ' ')
   387→    .split(' ')
   388→    .filter(Boolean)
   389→    .map(word =>
   390→      word
   391→        .toLowerCase()
   392→        .replace(/(^|-)([a-zа-яё])/gi, (_m, prefix: string, char: string) => prefix + char.toUpperCase())
   393→    )
   394→    .join(' ');
   395→}

   397→function numberToWords(n: number): string {
   398→  const units = ["", "один", "два", "три", "четыре", "пять", "шесть", "семь", "восемь", "девять"];
   399→  const teens = ["десять", "одиннадцать", "двенадцать", "тринадцать", "четырнадцать", "пятнадцать", "шестнадцать", "семнадцать", "восемнадцать", "девятнадцать"];
   400→  const tens = ["", "", "двадцать", "тридцать", "сорок", "пятьдесят", "шестьдесят", "семьдесят", "восемьдесят", "девяносто"];
   401→  const hundreds = ["", "сто", "двести", "триста", "четыреста", "пятьсот", "шестьсот", "семьсот", "восемьсот", "девятьсот"];
   402→
   403→  if (n >= 1000000) return String(n);
   404→  if (n >= 1000) {
   405→    const th = Math.floor(n / 1000);
   406→    const r = n % 1000;
   407→    const lastTwo = th % 100, lastOne = th % 10;
   408→    let w = "тысяч";
   409→    if (lastTwo >= 11 && lastTwo <= 19) w = "тысяч";
   410→    else if (lastOne === 1) w = "тысяча";
   411→    else if (lastOne >= 2 && lastOne <= 4) w = "тысячи";
   412→    return numberToWords(th) + " " + w + (r > 0 ? " " + numberToWords(r) : "");
   413→  }
   414→  if (n === 0) return "ноль";
   415→  if (n < 10) return units[n];
   416→  if (n < 20) return teens[n - 10];
   417→  if (n < 100) {
   418→    const t = Math.floor(n / 10), u = n % 10;
   419→    return tens[t] + (u > 0 ? " " + units[u] : "");
   420→  }
   421→  if (n < 1000) {
   422→    const h = Math.floor(n / 100), r = n % 100;
   423→    return hundreds[h] + (r > 0 ? " " + numberToWords(r) : "");
   424→  }
   425→  return String(n);
   426→}

   439→async function getState(userId: string): Promise<{ state: string; context: TestDriveContext } | null> {
   440→  const { data } = await supabaseAdmin
   441→    .from("user_states")
   442→    .select("state, context, expires_at")

   444→    .maybeSingle();

   483→    if (context.licenseCategory) vu += ` (${context.licenseCategory})`;

   486→    lines.push(`🚗 В/У: не указано`);

   489→  lines.push(
   490→    "",
   491→    "Всё верно?",

   494→  return lines.join("\n");

   924→
   925→    return true;
   926→  } catch (error) {
   927→    logger.error("[/testdrive] Generate failed", error);
   928→    await sendComplexMessage(chatId, "🚨 Ошибка. Попробуйте ещё раз.", [], { removeKeyboard: true });
   929→    return false;
   930→  }

   935→export async function handleTestDriveText(userId: string, chatId: number, text: string): Promise<boolean> {
   936→  const tdState = await getState(userId);
   937→  if (!tdState) return false;
   938→
   939→  const { state, context } = tdState;
   940→
   941→  // ── Phone ──
   942→  if (state === "td_phone") {
   943→    const cleaned = text.replace(/[^\d+]/g, "");
   944→    if (cleaned.length < 10) {
   945→      await sendComplexMessage(chatId, "❌ Неверный формат.\n\nПример: `+7 987 654 32 10`", [], { removeKeyboard: true, parseMode: "Markdown" });
   946→      return true;
   947→    }
   948→    context.customerPhone = cleaned;
   949→    await setState(userId, "td_doc_choice", context);
   950→    await sendComplexMessage(
   951→      chatId,
   952→      `✅ ${cleaned}\n\n*Какой документ вводим?*\n\nМожно ввести паспорт ИЛИ водительское удостоверение (достаточно одного).`,
   953→      buildDocChoiceKeyboard(),
   954→      { keyboardType: "inline", parseMode: "Markdown" },
   955→    );
   956→    return true;
   957→  }
   958→
   959→  // ── Doc choice (text fallback — re-prompt with keyboard) ──
   960→  if (state === "td_doc_choice") {
   961→    await sendComplexMessage(
   962→      chatId,
   963→      "*Какой документ вводим?*\n\nВыберите на клавиатуре:",
   964→      buildDocChoiceKeyboard(),
   965→      { keyboardType: "inline", parseMode: "Markdown" },
   966→    );
   967→    return true;
   968→  }
   969→
   970→  // ── Full name ──
   971→  if (state === "td_name") {
   972→    context.customerFullName = capitalizeFullName(text);
   973→    if (context.needPassport) {
   974→      await setState(userId, "td_passport", context);
   975→      await sendComplexMessage(
   976→        chatId,
   977→        `✅ ${text}\n\n*Паспорт*\n\n4509 123456 15.03.2020 ОМВД по Н.Новгороду`,
   978→        [],
   979→        { removeKeyboard: true, parseMode: "Markdown" },
   980→      );
   981→    } else {
   982→      // License-only path — skip passport, go straight to license input
   983→      await setState(userId, "td_license", context);
   984→      await sendComplexMessage(
   985→        chatId,
   986→        `✅ ${text}\n\n*Водительское удостоверение*\n\nФормат: серия номер [дата_выдачи] [дата_окончания]\n\nПримеры:\n• 99 76 123456 15.03 (срок auto +10 лет)\n• 99 76 123456 (без дат)`,
   987→        [],
   988→        { removeKeyboard: true, parseMode: "Markdown" },
   989→      );
   990→    }
   991→    return true;
   992→  }
   993→
   994→  // ── Passport ──
   995→  if (state === "td_passport") {
   996→    const p = parsePassport(text);
   997→    if (!p) {
   998→      await sendComplexMessage(chatId, "❌ Формат: 4509 123456 15.03.2020 ОМВД", [], { removeKeyboard: true });
   999→      return true;
  1000→    }
  1001→    context.customerSeries = p.series;
  1002→    context.customerNumber = p.number;
  1003→    context.customerIssueDate = p.issueDate;
  1004→    context.customerIssuedBy = p.issuedBy;
  1005→    await setState(userId, "td_birth", context);
  1006→    await sendComplexMessage(
  1007→      chatId,
  1008→      `✅ Паспорт ${p.series} ${p.number} от ${p.issueDate}\n\n*Дата рождения*\n\n15.03.1990`,
  1009→      [],
  1010→      { removeKeyboard: true, parseMode: "Markdown" },
  1011→    );
  1012→    return true;
  1013→  }
  1014→
  1015→  // ── Birth date ──
  1016→  if (state === "td_birth") {
  1017→    const d = parseDate(text, true);
  1018→    if (!d) {
  1019→      await sendComplexMessage(chatId, "❌ Формат: 15.03.1990", [], { removeKeyboard: true });
  1020→      return true;
  1021→    }
  1022→    context.customerBirthDate = d;
  1023→    await setState(userId, "td_address", context);
  1024→    await sendComplexMessage(chatId, `✅ ${d}\n\n*Адрес регистрации*\n\nПример: г. Нижний Новгород, ул. Ленина, д. 1, кв. 1`, [], { removeKeyboard: true, parseMode: "Markdown" });
  1025→    return true;
  1026→  }
  1027→
  1028→  // ── Address ──
  1029→  if (state === "td_address") {
  1030→    context.customerRegistration = text.trim();
  1031→    // After passport path: if license also needed → go to license; else → confirm
  1032→    if (context.needLicense) {
  1033→      await setState(userId, "td_license", context);
  1034→      await sendComplexMessage(
  1035→        chatId,
  1036→        `✅\n\n*Водительское удостоверение*\n\nФормат: серия номер [дата_выдачи] [дата_окончания]\n\nПримеры:\n• 99 76 123456 15.03 (срок auto +10 лет)\n• 99 76 123456 (без дат)`,
  1037→        [],
  1038→        { removeKeyboard: true, parseMode: "Markdown" },
  1039→      );
  1040→    } else {
  1041→      await gotoConfirm(chatId, userId, context);
  1042→    }
  1043→    return true;
  1044→  }
  1045→
  1046→  // ── License input ──
  1047→  if (state === "td_license") {
  1048→    const l = parseLicense(text);
  1049→    if (!l) {
  1050→      await sendComplexMessage(
  1051→        chatId,
  1052→        `❌ Не удалось распознать. Формат: серия номер [дата_выдачи] [дата_окончания]\n\nПримеры:\n• 99 76 123456 15.03 (срок auto +10 лет)\n• 9976 123456 15.03.2020 15.03.2030\n• 99 76 123456 (без дат)`,
  1053→        [],
  1054→        { removeKeyboard: true, parseMode: "Markdown" },
  1055→      );
  1056→      return true;
  1057→    }
  1058→    context.licenseSeries = l.series;
  1059→    context.licenseNumber = l.number;
  1060→    context.licenseIssueDate = l.issueDate || undefined;
  1061→    context.licenseExpiryDate = l.expiryDate || undefined;
  1062→    // After license → ask for categories
  1063→    await setState(userId, "td_category", context);
  1064→    const dateStr = l.issueDate
  1065→      ? ` от ${l.issueDate}${l.expiryDate ? ` до ${l.expiryDate}` : ""}`
  1066→      : "";
  1067→    await sendComplexMessage(
  1068→      chatId,
  1069→      `✅ В/У ${l.series} № ${l.number}${dateStr}\n\n*Категория ВУ*`,
  1070→      buildCategoryKeyboard(),
  1071→      { keyboardType: "inline", parseMode: "Markdown" },
  1072→    );
  1073→    return true;
  1074→  }
  1075→
  1076→  // ── Category (text fallback — re-prompt with keyboard) ──
  1077→  if (state === "td_category") {
  1078→    await sendComplexMessage(chatId, "*Категория ВУ*", buildCategoryKeyboard(), { keyboardType: "inline", parseMode: "Markdown" });
  1079→    return true;
  1080→  }
  1081→
  1082→  return false;
  1083→}

  1087→export async function handleTestDriveCallback(
  1088→  userId: string,
  1089→  chatId: number,
  1090→  callbackData: string,
  1091→  callbackQueryId?: string,
  1092→): Promise<boolean> {
  1093→  const tdState = await getState(userId);
  1094→  if (!tdState) return false;
  1095→
  1096→  const { state, context } = tdState;
  1097→
  1098→  // Resolve crew slug from user_states
  1099→  const crewSlug = await getTDCrewSlug(userId);
  1100→
  1101→  // Answer callback query to stop loading animation
  1102→  if (callbackQueryId) {
  1103→    try {
  1104→      await fetch(`https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/answerCallbackQuery?callback_query_id=${callbackQueryId}`, { method: "POST" });
  1105→    } catch (e) {
  1106→      logger.warn("[/testdrive] Failed to answer callback query:", e);
  1107→    }
  1108→  }
  1109→
  1110→  // ── Global actions ──
  1111→
  1112→  if (callbackData === "td_cancel") {
  1113→    await sendComplexMessage(chatId, "❌ Отменено. /testdrive для начала.", [], { removeKeyboard: true });
  1114→    await clearState(userId);
  1115→    return true;
  1116→  }
  1117→
  1118→  if (callbackData === "td_restart") {
  1119→    await clearState(userId);
  1120→    await testDriveCommand(chatId, parseInt(userId), undefined, "/testdrive");
  1121→    return true;
  1122→  }
  1123→
  1124→  // ── Bike selection (state: td_bike) ──
  1125→
  1126→  if (state === "td_bike" && callbackData.startsWith("td_bike_")) {
  1127→    const bikeId = callbackData.replace("td_bike_", "");
  1128→    const bike = await resolveBikeById(bikeId);
  1129→    if (!bike) {
  1130→      await sendComplexMessage(chatId, "🚲 Байк не найден. Попробуйте снова /testdrive", [], { removeKeyboard: true });
  1131→      await clearState(userId);
  1132→      return true;
  1133→    }
  1134→    context.bikeId = bike.id;
  1135→    context.bikeMake = bike.make;
  1136→    context.bikeModel = bike.model;
  1137→    await setState(userId, "td_phone", context);
  1138→    await sendComplexMessage(chatId, `🏍 ${bike.make} ${bike.model}`, [], { removeKeyboard: true });
  1139→    await sendComplexMessage(chatId, "*Телефон клиента*\nПример: `+7 987 654 32 10`", [], { parseMode: "Markdown", removeKeyboard: true });
  1140→    return true;
  1141→  }
  1142→
  1143→  // ── Document choice (state: td_doc_choice) ──
  1144→
  1145→  if (state === "td_doc_choice" && callbackData === "td_doc_passport") {
  1146→    context.needPassport = true;
  1147→    context.needLicense = false;
  1148→    await setState(userId, "td_name", context);
  1149→    await sendComplexMessage(chatId, "✅ Паспорт\n\n*ФИО*\n\nПример: `Иванов Иван Иванович`", [], { removeKeyboard: true, parseMode: "Markdown" });
  1150→    return true;
  1151→  }
  1152→
  1153→  if (state === "td_doc_choice" && callbackData === "td_doc_license") {
  1154→    context.needPassport = false;
  1155→    context.needLicense = true;
  1156→    await setState(userId, "td_name", context);
  1157→    await sendComplexMessage(chatId, "✅ В/У\n\n*ФИО*\n\nПример: `Иванов Иван Иванович`", [], { removeKeyboard: true, parseMode: "Markdown" });
  1158→    return true;
  1159→  }
  1160→
  1161→  if (state === "td_doc_choice" && callbackData === "td_doc_both") {
  1162→    context.needPassport = true;
  1163→    context.needLicense = true;
  1164→    await setState(userId, "td_name", context);
  1165→    await sendComplexMessage(chatId, "✅ Оба документа\n\n*ФИО*\n\nПример: `Иванов Иван Иванович`", [], { removeKeyboard: true, parseMode: "Markdown" });
  1166→    return true;
  1167→  }
  1168→
  1169→  // ── License category selection (state: td_category) ──
  1170→
  1171→  if (state === "td_category" && callbackData.startsWith("td_cat_")) {
  1172→    const cat = callbackData.replace("td_cat_", "");
  1173→    if (cat === "A" || cat === "B" || cat === "none") {
  1174→      const label = cat === "none" ? "нет" : cat;
  1175→      context.licenseCategory = label;
  1176→      await setState(userId, "td_category", context);
  1177→      await sendComplexMessage(chatId, `🏷 ${label}`, buildCategoryKeyboard(label), { keyboardType: "inline" });
  1178→    } else if (cat === "done") {
  1179→      if (!context.licenseCategory) {
  1180→        await sendComplexMessage(chatId, "Выберите категорию:", buildCategoryKeyboard(), { keyboardType: "inline" });
  1181→        return true;
  1182→      }
  1183→      // Done → advance to confirm
  1184→      await gotoConfirm(chatId, userId, context);
  1185→    }
  1186→    return true;
  1187→  }
  1188→
  1189→  // ── Confirm ──
  1190→
  1191→  if (state === "td_confirm" && callbackData === "td_ok") {
  1192→    await sendComplexMessage(chatId, "⏳ Генерирую договор...", [], { removeKeyboard: true });
  1193→    const success = await generateContract(chatId, userId, context, crewSlug);
  1194→    if (success) {
  1195→      await clearState(userId);
  1196→    }
  1197→    return true;
  1198→  }
  1199→
  1200→  return false;
  1201→}

  1205→export async function testDriveCommand(
  1206→  chatId: number,
  1207→  userId: number,
  1208→  username: string | undefined,
  1209→  text: string,
  1210→) {

  1212→  logger.info(`[/testdrive] ${userIdStr}: ${text}`);
  1213→

  1217→  // Show inline bike selection

  1219→  if (!bikes.length) {
  1220→    await sendComplexMessage(chatId, "🚲 Нет байков.", [], { removeKeyboard: true });

  1225→    const tier = b.specs?.access_tier;
  1226→    const emoji = tier === "pro" ? "🔴" : tier === "mid" ? "🟡" : tier === "entry" ? "🟢" : "⚪";

  1230→
  1231→  await setState(userIdStr, "td_bike", { bikeId: "" });
  1232→  await sendComplexMessage(
  1233→    chatId,
  1234→    "🛵 *Тест-драйв — выберите байк*",
  1235→    buttons,
  1236→    { keyboardType: "inline", parseMode: "Markdown" },
  1237→  );
  1238→}
