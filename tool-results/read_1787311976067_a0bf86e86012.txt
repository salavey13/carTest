   122→   122→async function resolveBikeById(bikeId: string): Promise<any> {
   123→   123→  const { data: exactMatch } = await supabaseAdmin
   124→   124→    .from("cars")
   125→   125→    .select("id, make, model, specs, type")
   126→   126→    .eq("id", bikeId)
   127→   127→    .in("type", ["bike", "ebike"])
   128→   128→    .maybeSingle();
   129→   129→  if (exactMatch) return exactMatch;
   130→   130→
   131→   131→  const { data: candidates } = await supabaseAdmin
   132→   132→    .from("cars")
   133→   133→    .select("id, make, model, specs, type")
   134→   134→    .in("type", ["bike", "ebike"])
   135→   135→    .limit(100);
   136→   136→  if (!candidates?.length) return null;
   137→   137→
   138→   138→  const norm = (v = "") => String(v).toLowerCase().replace(/[^a-zа-я0-9]+/gi, " ").trim();
   139→   139→  const qn = norm(bikeId);
   140→   140→  if (!qn) return null;
   141→   141→
   142→   142→  let best: any = null, bestScore = 0;
   143→   143→  for (const bike of candidates) {
   144→   144→    const hay = [bike.id, bike.make, bike.model, bike.specs?.vin, bike.specs?.frame].map(norm).join(" ");
   145→   145→    if (hay.includes(qn)) { best = bike; bestScore = 1000; break; }
   146→   146→    const parts = qn.split(" ");
   147→   147→    let score = 0;
   148→   148→    for (const p of parts) if (p && hay.includes(p)) score += 20 + p.length;
   149→   149→    if (score > bestScore) { bestScore = score; best = bike; }
   150→   150→  }
   151→   151→  return bestScore > 0 ? best : null;
   152→   152→}

   223→   223→function parsePassport(text: string): { series: string; number: string; issueDate: string; issuedBy: string } | null {
   224→   224→  const parts = text.trim().split(/\s+/);
   225→   225→  if (parts.length < 3) return null;
   226→   226→
   227→   227→  let series = "", number = "", dateIdx = -1;
   228→   228→  for (let i = 0; i < parts.length; i++) {
   229→   229→    const cleaned = parts[i].replace(/\D/g, '');
   230→   230→    if (!series && cleaned.length === 4) series = cleaned;
   231→   231→    else if (!number && cleaned.length === 6) number = cleaned;
   232→   232→    else if (dateIdx === -1 && /^\d{1,2}\.\d{1,2}(\.\d{2,4})?$/.test(parts[i])) dateIdx = i;
   233→   233→  }
   234→   234→
   235→   235→  if (!series || !number || dateIdx === -1) return null;
   236→   236→
   237→   237→  let dateStr = parts[dateIdx];
   238→   238→  const dateParts = dateStr.split('.');
   239→   239→  if (dateParts.length === 2) return null;
   240→   240→  if (dateParts[2].length === 2) {
   241→   241→    const y = parseInt(dateParts[2]);
   242→   242→    dateParts[2] = y > 50 ? `19${y}` : `20${y}`;
   243→   243→    dateStr = dateParts.join('.');
   244→   244→  }
   245→   245→
   246→   246→  const issuedBy = parts.slice(dateIdx + 1).join(' ') || "не указано";
   247→   247→  return { series, number, issueDate: dateStr, issuedBy };
   248→   248→}

   250→   250→/**
   251→   251→ * Parse Russian VU (driver's license) series + number + dates.
   252→   252→ * Ported from doc-manual.ts parseLicense() with testdrive's 4-strategy series parsing.
   253→   253→ *
   254→   254→ * Format: серия номер [дата_выдачи [дата_окончания]]
   255→   255→ *   "9976 123456 15.03"          → series=9976, num=123456, issue=15.03.2026, expiry=15.03.2036
   256→   256→ *   "99 76 123456 15.03.2020 15.03.2030"  → series=9976, num=123456, issue=15.03.2020, expiry=15.03.2030
   257→   257→ *   "99 76 123456"               → no dates → still returns series+number with null dates
   258→   258→ *   "99 76123456 15.03"          → Strategy 4: 2+8 split → series=9976, num=123456, issue=15.03.2026
   259→   259→ *
   260→   260→ * If only one date: it's treated as issue date, expiry = issue + 10 years.
   261→   261→ * Year defaults to CURRENT_YEAR if omitted.
   262→   262→ */
   263→   263→function parseLicense(text: string): { series: string; number: string; issueDate: string | null; expiryDate: string | null } | null {
   264→   264→  const parts = text.trim().split(/\s+/).filter(Boolean);
   265→   265→  if (parts.length < 2) return null;
   266→   266→
   267→   267→  // Separate digit sequences from date-like tokens
   268→   268→  const digits: string[] = [];
   269→   269→  const dates: string[] = [];
   270→   270→
   271→   271→  for (const p of parts) {
   272→   272→    if (/^\d{1,2}\.\d{1,2}(\.\d{2,4})?$/.test(p)) {
   273→   273→      // It's a date-like token
   274→   274→      let d = p;
   275→   275→      if (!d.includes('.')) continue;
   276→   276→      const dparts = d.split('.');
   277→   277→      if (dparts.length === 2) d += `.${CURRENT_YEAR}`;
   278→   278→      else if (dparts[2].length === 2) {
   279→   279→        const y = parseInt(dparts[2]);
   280→   280→        dparts[2] = y > 50 ? `19${y}` : `20${y}`;
   281→   281→        d = dparts.join('.');
   282→   282→      }
   283→   283→      dates.push(d);
   284→   284→    } else {
   285→   285→      const cleaned = p.replace(/\D/g, '');
   286→   286→      if (cleaned.length > 0) digits.push(cleaned);
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  if (digits.length < 2) return null;
   291→   291→
   292→   292→  let series = "", number = "";
   293→   293→
   294→   294→  // Strategy 1: find a 4-digit (series) followed by a 6-digit (number)
   295→   295→  for (let i = 0; i < digits.length; i++) {
   296→   296→    if (digits[i].length === 4) {
   297→   297→      series = digits[i];
   298→   298→      for (let j = i + 1; j < digits.length; j++) {
   299→   299→        if (digits[j].length === 6) { number = digits[j]; break; }
   300→   300→      }
   301→   301→      if (number) break;
   302→   302→    }
   303→   303→  }
   304→   304→
   305→   305→  // Strategy 2: two consecutive 2-digit tokens = series (e.g. "99 76"), then 6-digit = number
   306→   306→  if (!series || !number) {
   307→   307→    for (let i = 0; i < digits.length - 1; i++) {
   308→   308→      if (digits[i].length === 2 && digits[i + 1].length === 2) {
   309→   309→        series = digits[i] + digits[i + 1];
   310→   310→        for (let j = i + 2; j < digits.length; j++) {
   311→   311→          if (digits[j].length === 6) { number = digits[j]; break; }
   312→   312→        }
   313→   313→        if (number) break;
   314→   314→      }
   315→   315→    }
   316→   316→  }
   317→   317→
   318→   318→  // Strategy 3: find any 4+ digit for series, any 6-digit for number
   319→   319→  if (!series || !number) {
   320→   320→    for (const d of digits) {
   321→   321→      if (!series && d.length >= 4) series = d.slice(0, 4);
   322→   322→      else if (!number && d.length === 6) number = d;
   323→   323→    }
   324→   324→  }
   325→   325→
   326→   326→  // Strategy 4: "99 76123456" -> 2-char + 8-char (where 8 = 2+6 concatenated)
   327→   327→  if (!series || !number) {
   328→   328→    for (let i = 0; i < digits.length - 1; i++) {
   329→   329→      if (digits[i].length === 2 && digits[i + 1].length === 8) {
   330→   330→        series = digits[i] + digits[i + 1].slice(0, 2);
   331→   331→        number = digits[i + 1].slice(2);
   332→   332→        if (number.length === 6) break;
   333→   333→        series = ""; number = "";
   334→   334→      }
   335→   335→    }
   336→   336→  }
   337→   337→
   338→   338→  if (!series || !number) return null;
   339→   339→
   340→   340→  // ── Date handling (ported from doc-manual.ts) ──
   341→   341→  let issueDate: string | null = null;
   342→   342→  let expiryDate: string | null = null;
   343→   343→
   344→   344→  if (dates.length === 1) {
   345→   345→    // Single date → issue date, expiry = +10 years
   346→   346→    issueDate = dates[0];
   347→   347→    const [dd, mm, yyyy] = issueDate.split('.');
   348→   348→    expiryDate = `${dd}.${mm}.${parseInt(yyyy) + 10}`;
   349→   349→  } else if (dates.length >= 2) {
   350→   350→    // Two dates → issue and expiry
   351→   351→    issueDate = dates[0];
   352→   352→    expiryDate = dates[1];
   353→   353→    // If both same year → auto extend (e.g., "15.03.2026 15.03.2026" → +10y)
   354→   354→    if (issueDate === expiryDate && issueDate.endsWith(`.${CURRENT_YEAR}`)) {
   355→   355→      const [dd, mm, yyyy] = issueDate.split('.');
   356→   356→      expiryDate = `${dd}.${mm}.${parseInt(yyyy) + 10}`;
   357→   357→    }
   358→   358→  }
   359→   359→  // If no dates → issueDate/expiryDate stay null (template {{#if}} hides them)
   360→   360→
   361→   361→  return { series, number, issueDate, expiryDate };
   362→   362→}

   364→   364→function parseDate(text: string, requireYear = true): string | null {
   365→   365→  const match = text.trim().match(/^(\d{1,2})\.(\d{1,2})(\.(\d{2,4}))?$/);
   366→   366→  if (!match) return null;
   367→   367→
   368→   368→  let [, day, month, , year] = match;
   369→   369→  day = day.padStart(2, '0');
   370→   370→  month = month.padStart(2, '0');
   371→   371→
   372→   372→  if (!year) {
   373→   373→    if (requireYear) return null;
   374→   374→    year = String(CURRENT_YEAR);
   375→   375→  } else if (year.length === 2) {
   376→   376→    const y = parseInt(year);
   377→   377→    year = y > 50 ? `19${y}` : `20${y}`;
   378→   378→  }
   379→   379→
   380→   380→  return `${day}.${month}.${year}`;
   381→   381→}

   397→   397→function numberToWords(n: number): string {
   398→   398→  const units = ["", "один", "два", "три", "четыре", "пять", "шесть", "семь", "восемь", "девять"];
   399→   399→  const teens = ["десять", "одиннадцать", "двенадцать", "тринадцать", "четырнадцать", "пятнадцать", "шестнадцать", "семнадцать", "восемнадцать", "девятнадцать"];
   400→   400→  const tens = ["", "", "двадцать", "тридцать", "сорок", "пятьдесят", "шестьдесят", "семьдесят", "восемьдесят", "девяносто"];
   401→   401→  const hundreds = ["", "сто", "двести", "триста", "четыреста", "пятьсот", "шестьсот", "семьсот", "восемьсот", "девятьсот"];
   402→   402→
   403→   403→  if (n >= 1000000) return String(n);
   404→   404→  if (n >= 1000) {
   405→   405→    const th = Math.floor(n / 1000);
   406→   406→    const r = n % 1000;
   407→   407→    const lastTwo = th % 100, lastOne = th % 10;
   408→   408→    let w = "тысяч";
   409→   409→    if (lastTwo >= 11 && lastTwo <= 19) w = "тысяч";
   410→   410→    else if (lastOne === 1) w = "тысяча";
   411→   411→    else if (lastOne >= 2 && lastOne <= 4) w = "тысячи";
   412→   412→    return numberToWords(th) + " " + w + (r > 0 ? " " + numberToWords(r) : "");
   413→   413→  }
   414→   414→  if (n === 0) return "ноль";
   415→   415→  if (n < 10) return units[n];
   416→   416→  if (n < 20) return teens[n - 10];
   417→   417→  if (n < 100) {
   418→   418→    const t = Math.floor(n / 10), u = n % 10;
   419→   419→    return tens[t] + (u > 0 ? " " + units[u] : "");
   420→   420→  }
   421→   421→  if (n < 1000) {
   422→   422→    const h = Math.floor(n / 100), r = n % 100;
   423→   423→    return hundreds[h] + (r > 0 ? " " + numberToWords(r) : "");
   424→   424→  }
   425→   425→  return String(n);
   426→   426→}

  1177→  1177→      await sendComplexMessage(chatId, `🏷 ${label}`, buildCategoryKeyboard(label), { keyboardType: "inline" });
  1178→  1178→    } else if (cat === "done") {
  1179→  1179→      if (!context.licenseCategory) {
  1180→  1180→        await sendComplexMessage(chatId, "Выберите категорию:", buildCategoryKeyboard(), { keyboardType: "inline" });

  1182→  1182→      }
  1183→  1183→      // Done → advance to confirm
  1184→  1184→      await gotoConfirm(chatId, userId, context);
  1185→  1185→    }
  1186→  1186→    return true;
  1187→  1187→  }
  1188→  1188→
  1189→  1189→  // ── Confirm ──
  1190→  1190→
  1191→  1191→  if (state === "td_confirm" && callbackData === "td_ok") {
  1192→  1192→    await sendComplexMessage(chatId, "⏳ Генерирую договор...", [], { removeKeyboard: true });
  1193→  1193→    const success = await generateContract(chatId, userId, context, crewSlug);
  1194→  1194→    if (success) {
  1195→  1195→      await clearState(userId);
  1196→  1196→    }
  1197→  1197→    return true;
  1198→  1198→  }
  1199→  1199→
  1200→  1200→  return false;
  1201→  1201→}
