    12→// ── Auth helper ─────────────────────────────────────────────────────────────
    13→// Verifies that the caller has access to the given crew.
    14→// Uses TWO verification paths (both server-side, no client-supplied booleans):
    15→//
    16→// 1. Telegram WebApp: reads TELEGRAM_ACTOR_COOKIE (signed HMAC-SHA256) → gets real userId
    17→//    → checks crew owner / admin / active crew member
    18→// 2. Password auth: caller passes actorUserId (the crew owner's ID from validateAnalyticsPassword)
    19→//    → server verifies actorUserId === crew.owner_id (only the real owner would know this UUID)
    20→//
    21→// LA-001 FIX: was trusting a client-supplied isPasswordAuth boolean — anyone could bypass
    22→// auth by passing isPasswordAuth=true. Now the server verifies identity in both paths.
    23→async function verifyCrewAccess(
    24→  crewId: string,
    25→): Promise<{ allowed: boolean; actorUserId?: string; error?: string }> {
    26→  // Dynamic imports to avoid module-level server-only chain
    27→  const { cookies } = await import("next/headers");
    28→  const { TELEGRAM_ACTOR_COOKIE, verifyTelegramActorCookieValue } = await import("@/lib/telegram-actor-cookie");
    29→
    30→  // Path 1: Telegram WebApp — read signed cookie
    31→  const cookieUserId = verifyTelegramActorCookieValue(
    32→    (await cookies()).get(TELEGRAM_ACTOR_COOKIE)?.value,
    33→  );
    34→
    35→  if (cookieUserId) {
    36→    // Telegram auth — verify crew access via DB
    37→    const { data: user, error: userErr } = await supabaseAdmin
    38→      .from("users")
    39→      .select("metadata")
    40→      .eq("user_id", cookieUserId)
    41→      .maybeSingle();
    42→
    43→    if (userErr) logger.warn("[verifyCrewAccess] users query failed:", userErr.message);
    44→
    45→    const userMetadata = user?.metadata as Record<string, unknown> | null;
    46→    const isAdmin = userMetadata?.role === "admin" || userMetadata?.status === "admin";
    47→
    48→    // Check crew owner
    49→    const { data: crew, error: crewErr } = await supabaseAdmin
    50→      .from("crews")
    51→      .select("owner_id")
    52→      .eq("id", crewId)
    53→      .maybeSingle();
    54→
    55→    if (crewErr) logger.warn("[verifyCrewAccess] crews query failed:", crewErr.message);
    56→    const isOwner = crew?.owner_id === cookieUserId;
    57→
    58→    // Check crew member (LR3-006 FIX: filter by membership_status='active' — was missing,
    59→    // so removed/inactive members retained access indefinitely)
    60→    const { data: crewMember, error: memberErr } = await supabaseAdmin
    61→      .from("crew_members")
    62→      .select("user_id")
    63→      .eq("crew_id", crewId)
    64→      .eq("user_id", cookieUserId)
    65→      .eq("membership_status", "active")
    66→      .maybeSingle();
    67→
    68→    if (memberErr) logger.warn("[verifyCrewAccess] crew_members query failed:", memberErr.message);
    69→    const isCrewMember = !!crewMember;
    70→
    71→    if (isOwner || isAdmin || isCrewMember) {
    72→      return { allowed: true, actorUserId: cookieUserId };
    73→    }
    74→    return { allowed: false, error: "Недостаточно прав для просмотра данных этого экипажа." };
    75→  }
    76→
    77→  // Path 2: No Telegram cookie — not authenticated via WebApp
    78→  // (Password auth is handled by the caller passing actorUserId, which is verified
    79→  //  by checking actorUserId === crew.owner_id. This is done in the calling function
    80→  //  because it needs the crew owner_id which is fetched there.)
    81→  return { allowed: false, error: "Не авторизован." };
    82→}

    84→// ── Auth helper for password-authenticated users ────────────────────────────
    85→// Verifies that the provided actorUserId is the OWNER of the given crew.
    86→// This is the password-auth path: validateAnalyticsPassword returns the crew owner's
    87→// ID, and only the real owner would know this UUID. We verify ownership server-side.
    88→async function verifyCrewOwnerAccess(
    89→  actorUserId: string,
    90→  crewId: string,
    91→): Promise<{ allowed: boolean; error?: string }> {
    92→  const { data: crew, error: crewErr } = await supabaseAdmin
    93→    .from("crews")
    94→    .select("owner_id")
    95→    .eq("id", crewId)
    96→    .maybeSingle();
    97→
    98→  if (crewErr) logger.warn("[verifyCrewOwnerAccess] crews query failed:", crewErr.message);
    99→  if (!crew) return { allowed: false, error: "Экипаж не найден." };
   100→
   101→  if (crew.owner_id === actorUserId) {
   102→    return { allowed: true };
   103→  }
   104→
   105→  // Also check if actorUserId is a global admin (in case the password was
   106→  // generated by an admin for a crew they don't own)
   107→  const { data: user } = await supabaseAdmin
   108→    .from("users")
   109→    .select("metadata")
   110→    .eq("user_id", actorUserId)
   111→    .maybeSingle();
   112→
   113→  const userMetadata = user?.metadata as Record<string, unknown> | null;
   114→  if (userMetadata?.role === "admin" || userMetadata?.status === "admin") {
   115→    return { allowed: true };
   116→  }
   117→
   118→  return { allowed: false, error: "Недостаточно прав для просмотра данных этого экипажа." };
   119→}

   126→/**
   127→ * Fetch the set of operator Telegram IDs for a crew (owner + active members),
   128→ * and return the crew id so callers don't need a separate lookup.
   129→ *
   130→ * BUG FIX (previously): this function selected only `owner_id` from `crews`,
   131→ * then queried `crew_members` with `crew?.id` which was always `undefined` —
   132→ * so only the owner was ever detected as an operator. Active members who are
   133→ * co-owners/admins/mechanics were treated as real renters.
   134→ */
   135→async function getCrewOperatorIds(
   136→  slug: string,
   137→): Promise<{ ids: Set<string>; crewId: string | null; ownerId: string | null }> {
   138→  const ids = new Set<string>();
   139→  try {
   140→    const { data: crew } = await supabaseAdmin
   141→      .from("crews")
   142→      .select("id, owner_id")
   143→      .eq("slug", slug)
   144→      .maybeSingle();
   145→
   146→    if (crew?.owner_id) ids.add(crew.owner_id);
   147→    if (crew?.id) {
   148→      const { data: members } = await supabaseAdmin
   149→        .from("crew_members")
   150→        .select("user_id")
   151→        .eq("crew_id", crew.id)
   152→        .eq("membership_status", "active");
   153→
   154→      if (members) {
   155→        for (const m of members) {
   156→          if (m.user_id) ids.add(m.user_id);
   157→        }
   158→      }
   159→    }
   160→    return { ids, crewId: crew?.id ?? null, ownerId: crew?.owner_id ?? null };
   161→  } catch (error) {
   162→    logger.warn("[getCrewOperatorIds] Failed to fetch crew operators:", error);
   163→    return { ids, crewId: null, ownerId: null };
   164→  }
   165→}

   167→/** Check if a chat ID is a numeric Telegram ID (not a phone number or operator). */
   168→function isNumericTelegramId(id: string | null | undefined): boolean {
   169→  if (!id) return false;
   170→  // Telegram IDs are up to 10 digits today; allow up to 12 for future-proofing.
   171→  return /^\d{1,12}$/.test(id);
   172→}

   174→/** Check if a string looks like a phone number. */
   175→function isPhoneString(id: string | null | undefined): boolean {
   176→  if (!id) return false;
   177→  return /^(\+7|8|7)\d{10}$/.test(id.replace(/[\s\-\(\)]/g, ""));
   178→}

   180→/**
   181→ * Build a stable per-renter identity key from a full name, e.g.
   182→ * "Рудометов Михаил Сергеевич." → "name:рудометов михаил сергеевич".
   183→ *
   184→ * WHY: when an operator creates a contract via /doc-manual and SKIPS the optional
   185→ * client phone, the lead used to be keyed by the operator's telegram_chat_id.
   186→ * That collapsed ALL of one operator's renters into a single franchize_intents /
   187→ * lead row (last write wins on name), so only ONE renter was ever visible on the
   188→ * leads page and the other 44+ were invisible. Keying by normalized name instead
   189→ * gives every renter a distinct, stable lead identity even without a phone.
   190→ *
   191→ * Normalization is intentionally minimal (lowercase, strip trailing dots/punct,
   192→ * collapse whitespace) and MUST stay identical here and in leads-query.mjs so the
   193→ * web page and the text skill surface the same identities. Returns "" for empty.
   194→ */
   195→function nameIdentityKey(fullName: string | null | undefined): string {
   196→  const n = (fullName || "")
   197→    .trim()
   198→    .toLowerCase()
   199→    .replace(/[.\s]+/g, " ")
   200→    .replace(/[.]/g, "")
   201→    .trim();
   202→  return n ? `name:${n}` : "";
   203→}

   205→/**
   206→ * Normalize a phone number to canonical E.164-ish form (+7XXXXXXXXXX for RU).
   207→ * Accepts +7/7/8 prefix, spaces, dashes, parentheses.
   208→ * Returns null if input is empty or unparseable.
   209→ *
   210→ * BUG FIX: previously, leads were keyed by the raw phone string as typed by the
   211→ * operator in /doc-manual ("8 999 123-45-67"), while todos were keyed by the
   212→ * raw leadPhone passed to createLeadFollowupTodos ("+79991234567").
   213→ * Same person → two different lead cards. Normalizing at every read & write
   214→ * path collapses them into one canonical identity.
   215→ */
   216→/**
   217→ * Classify identity state for a lead.
   218→ *
   219→ * BUG FIX: previously, after QR claim overwrote telegram_user_id with the renter's
   220→ * id, the operator origin was lost and the lead was misclassified as 'claimed_user'.
   221→ * Now we consult `originalOperatorChatId` (preserved across QR claim from
   222→ * rentals.created_by_operator_chat_id, rental_contract_artifacts.created_by_operator_chat_id,
   223→ * or franchize_intents.metadata.operatorId) to detect 'merged' state — operator
   224→ * created it, renter claimed it.
   225→ */
   226→function classifyIdentityState(
   227→  lead: { user_id: string; phone: string | null; telegramChatId?: string | null; sourceCount?: number; originalOperatorChatId?: string | null },
   228→  crewOperatorIds: Set<string>,
   229→): 'claimed_user' | 'phone_only' | 'operator_placeholder' | 'merged' {
   230→  const userId = lead.user_id;
   231→  const originalOp = lead.originalOperatorChatId || null;
   232→
   233→  // Operator placeholder: the visible id IS the operator (pre-claim state).
   234→  if (crewOperatorIds.has(userId)) return 'operator_placeholder';
   235→
   236→  // Merged: original creator was an operator, but the visible id is now someone else
   237→  // (post-QR-claim). Treat as merged so the UI can show "originally operator-placeholder".
   238→  if (originalOp && crewOperatorIds.has(originalOp) && originalOp !== userId) {
   239→    return 'merged';
   240→  }
   241→
   242→  // Check if identity is a phone number (no Telegram user linked)
   243→  if (isPhoneString(userId) || (lead.phone && userId === lead.phone)) return 'phone_only';
   244→
   245→  // Has a real Telegram user_id AND it's not an operator
   246→  if (isNumericTelegramId(userId)) return 'claimed_user';
   247→
   248→  // If merged from multiple source types
   249→  if ((lead.sourceCount || 0) >= 2) return 'merged';
   250→
   251→  // Fallback: look at telegramChatId
   252→  if (lead.telegramChatId && isNumericTelegramId(lead.telegramChatId) && !crewOperatorIds.has(lead.telegramChatId)) {
   253→    return 'claimed_user';
   254→  }
   255→
   256→  // Default: phone-only by elimination
   257→  if (lead.phone) return 'phone_only';
   258→
   259→  return 'operator_placeholder';
   260→}

   557→        // owns it, renter hasn't scanned QR yet).

   559→        // /doc-manual L1614-1615 sets:
   560→        //   telegram_chat_id = String(userId)              // operator's TG id
   561→        //   created_by_operator_chat_id = String(userId)   // same, preserved forever

   564→        //
   565→        // So: telegram_chat_id === created_by_operator_chat_id ⟺ pre-claim.

   567→        // operators who are no longer in crew_members, were never added, or were

   861→    // ── Parallel enrichment (steps 7-9, 11) ──
   862→    //
   863→    // BUG FIX (previously): three more silent failures here:
   864→    //   - public.users select included `phone` which is NOT a column (it's in metadata->>phone).
   865→    //     The whole query 400'd → no telegram username/full_name enrichment ever happened.
   866→    //   - crew_todos was filtered to category=lead_followup only, dropping rental_verification
   867→    //     todos (passport check, return checklist) which are tied to the same rental_id.
   868→    // Bike titles are now pre-fetched BEFORE the artifact/sale loops (see above), so they
   869→    // are no longer part of this enrichment batch.
   870→    const allUserIds = Array.from(leadMap.keys()).filter((id) => /^\d+$/.test(id));
   871→    const leadPhones = Array.from(leadMap.values()).map((l) => l.phone).filter(Boolean) as string[];
   872→
   873→    // privateSchema already dynamically imported at line 270 (same function scope)
   874→    const [tgUsersResult, secretByPhoneResult, troubledUsersResult, todosResult] = await Promise.all([
   875→      // 7. Enrich from public.users — drop the non-existent `phone` column; phone is in metadata.
   876→      allUserIds.length > 0
   877→        ? supabaseAdmin.from("users").select("user_id, username, full_name, metadata").in("user_id", allUserIds)
   878→        : { data: [], error: null },
   879→      // 8. Enrich from secrets by phone (crew-filtered)
   880→      leadPhones.length > 0
   881→        ? privateSchema().from("user_rental_secrets").select("chat_id, renter_phone").eq("crew_slug", safeSlug).in("renter_phone", leadPhones) as Promise<{ data: Array<{ chat_id: string | null; renter_phone: string | null }> | null }>
   882→        : { data: [], error: null },
   883→      // 9. Troubled users
   884→      supabaseAdmin.from("users").select("user_id, metadata").not("metadata->>troubled", "is", null),
   885→      // 11. Lead-linked todos (filtered by crew_id on DB side).
   886→      // BUG FIX: include both `lead_followup` AND `rental_verification` — the latter
   887→      // covers passport/odometer/return-checklist todos which are tied to the same
   888→      // rental_id and absolutely belong on the lead's card.
   889→      supabaseAdmin
   890→        .from("crew_todos")
   891→        .select("id, lead_id, user_id, phone, rental_id, title, description, status, priority, category, created_at, completed_at, assigned_to, due_date")
   892→        .eq("crew_id", crewId)
   893→        .in("category", ["lead_followup", "rental_verification"])
   894→        .order("created_at", { ascending: false }),
   895→    ]);
   896→
   897→    // Surface enrichment query errors so future bugs are visible.
   898→    if (tgUsersResult.error) logger.error("[getFranchizeLeads] users enrichment query failed:", tgUsersResult.error);
   899→    if (secretByPhoneResult.error) logger.error("[getFranchizeLeads] secrets-by-phone enrichment query failed:", secretByPhoneResult.error);
   900→    if (troubledUsersResult.error) logger.error("[getFranchizeLeads] troubled-users query failed:", troubledUsersResult.error);
   901→    if (todosResult.error) logger.error("[getFranchizeLeads] crew_todos query failed:", todosResult.error);
   902→
   903→    const tgUsers = tgUsersResult.data;
   904→    const secretByPhone = secretByPhoneResult.data;
   905→    const troubledUsers = troubledUsersResult.data;
   906→    const todos = todosResult.data;
   907→
   908→    // Backfill top-level bikeTitle on leads that didn't get one set during the
   909→    // artifact/sale step (e.g. leads whose only source is franchize_intents or
   910→    // user_rental_secrets). bikeTitleMap is already fully populated from the
   911→    // pre-fetch above, but rentals/sales rows already carry their own bikeTitle.
   912→    for (const l of leadMap.values()) {
   913→      if (!l.bikeTitle) {
   914→        const firstRentalWithTitle = l.rentals.find((r) => r.bikeTitle);
   915→        const firstSaleWithTitle = l.sales.find((s) => s.bikeTitle);
   916→        if (firstRentalWithTitle) l.bikeTitle = firstRentalWithTitle.bikeTitle;
   917→        else if (firstSaleWithTitle) l.bikeTitle = firstSaleWithTitle.bikeTitle;
   918→      }
   919→    }
   920→
   921→    // 7. Enrich telegramChatId and username from public.users.
   922→    // Phone is read from metadata->>phone (the users table has no phone column).
   923→    if (tgUsers) {
   924→      for (const l of leadMap.values()) {
   925→        const match = tgUsers.find((u) => u.user_id === l.user_id);
   926→        if (match) {
   927→          l.telegramChatId = l.user_id;
   928→          if (match.username && !l.username) l.username = match.username;
   929→          if (match.full_name && !l.full_name) l.full_name = match.full_name;
   930→          const meta = match.metadata as Record<string, unknown> | null;
   931→          const metaPhone = normalizePhone(meta?.phone as string | undefined);
   932→          if (metaPhone && !l.phone) l.phone = metaPhone;
   933→        }
   934→      }
   935→    }
   936→
   937→    // 8. Enrich telegramChatId from secrets by phone match
   938→    if (secretByPhone) {
   939→      for (const l of leadMap.values()) {
   940→        if (!l.telegramChatId && l.phone) {
   941→          // Compare normalized phones so "8 999..." in secrets matches "+7999..." in leads.
   942→          const match = secretByPhone.find((s) =>
   943→            s.chat_id &&
   944→            s.renter_phone &&
   945→            normalizePhone(s.renter_phone) === l.phone,
   946→          );
   947→          if (match) l.telegramChatId = match.chat_id;
   948→        }
   949→      }
   950→    }
   951→
   952→    // 9. Troubled users
   953→    const troubledMap = new Map<string, string | null>();
   954→    if (troubledUsers) {
   955→      for (const u of troubledUsers) {
   956→        const meta = u.metadata as Record<string, unknown> | null;
   957→        if (meta?.troubled === true) {
   958→          troubledMap.set(u.user_id, (meta.troubled_reason as string) || null);
   959→        }
   960→      }
   961→    }
   962→    for (const l of leadMap.values()) {
   963→      if (troubledMap.has(l.user_id)) {
   964→        l.troubled = true;
   965→        l.troubledReason = troubledMap.get(l.user_id) || null;
   966→      }
   967→    }
   968→
   969→    // 10. Aggregate totals and refs
   970→    for (const l of leadMap.values()) {
   971→      l.contractCount = l.rentals.length;
   972→      l.saleCount = l.sales.length;
   973→      const rentalTotal = l.rentals.reduce((s, r) => s + (Number(r.totalCost) || 0), 0);
   974→      const saleTotal = l.sales.reduce((s, sale) => s + (Number(sale.salePrice) || 0), 0);
   975→      l.totalSpent = rentalTotal + saleTotal;
   976→      const lastRental = l.rentals
   977→        .filter((r) => r.startDate)
   978→        .sort((a, b) => new Date(b.startDate || 0).getTime() - new Date(a.startDate || 0).getTime())[0];
   979→      l.lastRentalDate = lastRental?.startDate || null;
   980→      l.contractRef = lastRental?.rentalId || l.sales[0]?.saleId || null;
   981→    }
   982→
   983→    // ── Identity state classification ──
   984→    // Classify each lead so the UI can show operator placeholders, phone-only leads, etc.
   985→    for (const l of leadMap.values()) {
   986→      l.identityState = classifyIdentityState(l, crewOperatorIds);
   987→      // Compute pipeline stage, QR status, next action
   988→      l.stageKey = computeLeadStage(l);
   989→      l.qrStatus = computeQrStatus(l);
   990→      l.nextAction = (STAGE_NEXT_ACTION as Record<string, string>)[l.stageKey] || null;
   991→      l.ownerId = l.originalOperatorChatId || null;
   992→    }
   993→
   994→    // ── Filter out pure operator-placeholder leads with no activity ──
   995→    // If a lead's only source is operator artifacts with no renter phone, no rentals,
   996→    // no todos — it's noise. Keep leads with activity for operator visibility.
   997→    for (const [key, l] of leadMap.entries()) {
   998→      if (
   999→        l.identityState === 'operator_placeholder' &&
  1000→        l.rentals.length === 0 &&
  1001→        l.sales.length === 0 &&
  1002→        (l.sourceCount || 0) <= 1
  1003→      ) {
  1004→        // Still keep — they'll be filtered on client if user picks "hide placeholders"
  1005→        // But flag them explicitly
  1006→      }
  1007→    }
  1008→
  1009→    // ── Server-side filtering: return todos matching loaded leads ──
  1010→    //
  1011→    // Matching strategy (Phase 1 identity fix):
  1012→    //  1. rental_id from description JSON → if leadMap has a lead whose rental includes this rental_id
  1013→    //  2. user_id column (canonical Telegram chat_id)
  1014→    //  3. phone column (phone-only leads)
  1015→    //  4. lead_id column (legacy)
  1016→    //  5. description JSON identity fields (legacy)
  1017→    //
  1018→    // This ensures operator-created todos (which have user_id=operator in description)
  1019→    // can still be matched by rental_id BEFORE the QR claim propagates the real renter ID.
  1020→    const leadUserIds = new Set(Array.from(leadMap.keys()));
  1021→
  1022→    // Build rental_id → lead user_id lookup for rental_id-based todo matching
  1023→    const rentalIdToLeadId = new Map<string, string>();
  1024→    for (const [leadId, lead] of leadMap.entries()) {
  1025→      for (const r of lead.rentals) {
  1026→        if (r.rentalId) rentalIdToLeadId.set(r.rentalId, leadId);
  1027→      }
  1028→    }
  1029→
  1030→    /**
  1031→     * Extract rental_id from todo (direct column preferred, fall back to description JSON).
  1032→     */
  1033→    const getTodoRentalId = (t: typeof todos[number]): string | null => {
  1034→      // 1. Direct rental_id column (Phase 3c — real FK, no parsing needed)
  1035→      if (t.rental_id) return t.rental_id;
  1036→      // 2. Legacy: parse from description JSON
  1037→      if (!t.description) return null;
  1038→      try {
  1039→        const desc = JSON.parse(t.description);
  1040→        if (desc.rental_id && typeof desc.rental_id === 'string') return desc.rental_id;
  1041→      } catch { /* ignore */ }
  1042→      return null;
  1043→    };
  1044→
  1045→    /**
  1046→     * Get the lead identifier from a todo.
  1047→     *
  1048→     * BUG FIX: previously used /^\d{1,9}$/ which rejects 10-digit Telegram IDs
  1049→     * (e.g. 7813830016, one of the operators in the audit). Most modern users have
  1050→     * 10-digit IDs, so this regex silently dropped their todos. Now allows up to 12 digits.
  1051→     * Also normalizes phones so a todo keyed by "8999..." matches a lead keyed by "+7999...".
  1052→     */
  1053→    const getTodoLeadId = (t: typeof todos[number]): string | null => {
  1054→      // 1. user_id column — canonical Telegram chat_id
  1055→      if (t.user_id && /^\d{1,12}$/.test(t.user_id)) return t.user_id;
  1056→      // 2. phone column — phone-only leads (normalize for cross-source matching)
  1057→      if (t.phone) {
  1058→        const normalized = normalizePhone(t.phone);
  1059→        if (normalized) return normalized;
  1060→      }
  1061→      // 3. lead_id column — legacy fallback
  1062→      if (t.lead_id) {
  1063→        if (/^\d{1,12}$/.test(t.lead_id)) return t.lead_id;
  1064→        // Phone-shaped lead_id (legacy) — normalize.
  1065→        const normalizedLead = normalizePhone(t.lead_id);
  1066→        if (normalizedLead) return normalizedLead;
  1067→        if (t.lead_id.includes('-')) return t.lead_id;
  1068→      }
  1069→      // 4. description JSON — legacy fallback
  1070→      if (t.description) {
  1071→        try {
  1072→          const desc = JSON.parse(t.description);
  1073→          if (desc.user_id && typeof desc.user_id === 'string' && /^\d{1,12}$/.test(desc.user_id)) return desc.user_id;
  1074→          if (desc.phone && typeof desc.phone === 'string') {
  1075→            const normalized = normalizePhone(desc.phone);
  1076→            if (normalized) return normalized;
  1077→          }
  1078→          if (desc.lead_id && typeof desc.lead_id === 'string') {
  1079→            if (/^\d{1,12}$/.test(desc.lead_id)) return desc.lead_id;
  1080→            const normalizedLead = normalizePhone(desc.lead_id);
  1081→            if (normalizedLead) return normalizedLead;
  1082→            if (desc.lead_id.includes('-')) return desc.lead_id;
  1083→          }
  1084→        } catch { /* ignore */ }
  1085→      }
  1086→      return null;
  1087→    };
  1088→
  1089→    const filteredTodos = (todos || []).filter((t) => {
  1090→      // 1. Match by rental_id from description (strongest link — works before QR claim)
  1091→      const todoRentalId = getTodoRentalId(t);
  1092→      if (todoRentalId && rentalIdToLeadId.has(todoRentalId)) return true;
  1093→
  1094→      // 2-5. Match by identity fallback chain
  1095→      const todoLeadId = getTodoLeadId(t);
  1096→      if (todoLeadId && leadUserIds.has(todoLeadId)) return true;
  1097→
  1098→      return false;
  1099→    });
  1100→
  1101→    // Deduplicate by todo id — each todo row should appear only once
  1102→    const seenTodoId = new Set<string>();
  1103→    const dedupedTodos: typeof filteredTodos = [];
  1104→    for (const t of filteredTodos) {
  1105→      if (t.id && seenTodoId.has(t.id)) continue;
  1106→      if (t.id) seenTodoId.add(t.id);
  1107→      // Fallback dedup by lead_id|title for todos without id
  1108→      const key = `${t.lead_id || '?'}|${t.title}`;
  1109→      if (seenTodoId.has(key)) continue;
  1110→      seenTodoId.add(key);
  1111→      dedupedTodos.push(t);
  1112→    }
  1113→
  1114→    // Compute assignee for each lead from its todos
  1115→    const assigneeIds = new Set<string>();
  1116→    for (const t of dedupedTodos) {
  1117→      if (t.assigned_to) assigneeIds.add(t.assigned_to);
  1118→    }
  1119→    const assigneeIdsList = Array.from(assigneeIds);
  1120→    if (assigneeIdsList.length > 0) {
  1121→      const { data: assigneeUsers } = await supabaseAdmin
  1122→        .from("users")
  1123→        .select("user_id, username, full_name")
  1124→        .in("user_id", assigneeIdsList);
  1125→      const assigneeMap = new Map<string, { username: string | null; full_name: string | null }>();
  1126→      for (const u of assigneeUsers ?? []) {
  1127→        assigneeMap.set(u.user_id, { username: u.username, full_name: u.full_name });
  1128→      }
  1129→      for (const l of leadMap.values()) {
  1130→        l.assigneeId = computeAssignee(l, dedupedTodos);
  1131→        if (l.assigneeId) {
  1132→          const a = assigneeMap.get(l.assigneeId);
  1133→          l.assigneeName = a?.full_name || a?.username || null;
  1134→        }
  1135→        if (l.ownerId) {
  1136→          const o = assigneeMap.get(l.ownerId) || (tgUsers as any[])?.find((u: any) => u.user_id === l.ownerId);
  1137→          l.ownerName = (o as any)?.full_name || (o as any)?.username || null;
  1138→        }
  1139→      }
  1140→    }
  1141→
  1142→    return {
  1143→      success: true,
  1144→      leads: Array.from(leadMap.values()),
  1145→      todos: dedupedTodos.map((t) => ({ ...t, description: t.description })),
  1146→    };
  1147→  } catch (error) {
  1148→    logger.error("[getFranchizeLeads] failed:", error);
  1149→    return { success: false, error: "Не удалось загрузить лиды" };
  1150→  }
  1151→}

  1153→// ── Document Verification ──────────────────────────────────────────────────────
  1154→// Types (DocVerificationData, GetRentalDocVerificationResult) are re-exported
  1155→// from leads-types.ts at the top of this file.
  1156→
  1157→/**
  1158→ * Get document verification data for a rental.
  1159→ * Returns signed URLs for photos, OCR data from user_rental_secrets, and checklist status.
  1160→ * Auth: actorUserId must be the rental owner, a crew operator, or a global admin
  1161→ * (or authenticated via analytics password).
  1162→ */
  1163→export async function getRentalDocVerification(
  1164→  rentalId: string,
  1165→  actorUserId?: string,
  1166→  isPasswordAuth?: boolean,
  1167→): Promise<GetRentalDocVerificationResult> {
  1168→  const { privateSchema } = await import("@/lib/private-secrets");
  1169→  noStore();
  1170→  if (!rentalId) {
  1171→    return { success: false, error: "rentalId is required" };
  1172→  }
  1173→
  1174→  try {
  1175→    // 1. Fetch rental with photo paths and metadata
  1176→    const { data: rental, error: rentalError } = await supabaseAdmin
  1177→      .from("rentals")
  1178→      .select("rental_id, user_id, crew_id, owner_id, metadata, passport_mainpage_photo, passport_registration_photo, drivers_licence_frontal_photo")
  1179→      .eq("rental_id", rentalId)
  1180→      .single();
  1181→
  1182→    if (rentalError || !rental) {
  1183→      return { success: false, error: "Rental not found" };
  1184→    }
  1185→
  1186→    // ── Auth check (LA-002 FIX: no more isPasswordAuth boolean bypass) ──
  1187→    // Dynamic imports to avoid module-level server-only chain
  1188→    const { cookies } = await import("next/headers");
  1189→    const { TELEGRAM_ACTOR_COOKIE, verifyTelegramActorCookieValue } = await import("@/lib/telegram-actor-cookie");
  1190→    const cookieUserId = verifyTelegramActorCookieValue(
  1191→      (await cookies()).get(TELEGRAM_ACTOR_COOKIE)?.value,
  1192→    );
  1193→
  1194→    if (cookieUserId) {
  1195→      // Telegram auth — check if this is the rental owner or renter
  1196→      const isRentalOwner = rental.owner_id === cookieUserId;
  1197→      const isRenter = rental.user_id === cookieUserId;
  1198→
  1199→      if (!isRentalOwner && !isRenter && rental.crew_id) {
  1200→        // Not their rental — verify crew access
  1201→        const access = await verifyCrewAccess(rental.crew_id);
  1202→        if (!access.allowed) {
  1203→          return { success: false, error: access.error || "Недостаточно прав." };
  1204→        }
  1205→      } else if (!isRentalOwner && !isRenter) {
  1206→        return { success: false, error: "Недостаточно прав." };
  1207→      }
  1208→    } else if (actorUserId && isPasswordAuth) {
  1209→      // LR3-008 FIX: check rental ownership FIRST (handles null crew_id case).
  1210→      // Was: only checked crew_id, so rentals with null crew_id always failed
  1211→      // even for the legitimate owner.
  1212→      const isRentalOwner = rental.owner_id === actorUserId;
  1213→      const isRenter = rental.user_id === actorUserId;
  1214→
  1215→      if (isRentalOwner || isRenter) {
  1216→        // Owner/renter always has access — proceed
  1217→      } else if (rental.crew_id) {
  1218→        // Not the owner — verify crew ownership
  1219→        const ownerAccess = await verifyCrewOwnerAccess(actorUserId, rental.crew_id);
  1220→        if (!ownerAccess.allowed) {
  1221→          return { success: false, error: ownerAccess.error || "Недостаточно прав." };
  1222→        }
  1223→      } else {
  1224→        return { success: false, error: "Недостаточно прав." };
  1225→      }
  1226→    } else {
  1227→      // No valid auth
  1228→      return { success: false, error: "Не авторизован." };
  1229→    }
  1230→
  1231→    // 2. Generate signed URLs for photos (5 min expiry)
  1232→    const photoPaths = {
  1233→      passportMainpage: (rental as any).passport_mainpage_photo as string | null,
  1234→      passportRegistration: (rental as any).passport_registration_photo as string | null,
  1235→      driversLicence: (rental as any).drivers_licence_frontal_photo as string | null,
  1236→    };
  1237→
  1238→    const signedUrls: Record<string, string | null> = {};
  1239→    for (const [key, path] of Object.entries(photoPaths)) {
  1240→      if (path) {
  1241→        const { data: urlData, error: urlError } = await supabaseAdmin.storage
  1242→          .from("docpix")
  1243→          .createSignedUrl(path, 300); // 5 minutes
  1244→        if (urlError || !urlData?.signedUrl) {
  1245→          logger.warn(`[getRentalDocVerification] Failed to create signed URL for ${key}: ${urlError?.message || "no URL"}`);
  1246→          signedUrls[key] = null;
  1247→        } else {
  1248→          signedUrls[key] = urlData.signedUrl;
  1249→        }
  1250→      } else {
  1251→        signedUrls[key] = null;
  1252→      }
  1253→    }
  1254→
  1255→    // 3. Fetch OCR data from user_rental_secrets
  1256→    const { data: secrets } = await privateSchema()
  1257→      .from("user_rental_secrets")
  1258→      .select("renter_full_name, renter_passport, renter_passport_issued_by, renter_passport_issue_date, renter_birth_date, renter_registration, renter_driver_license")
  1259→      .eq("source_rental_id", rentalId)
  1260→      .maybeSingle();
  1261→
  1262→    // 4. Extract checklist from metadata
  1263→    const metadata = (rental.metadata || {}) as Record<string, any>;
  1264→    const checklist = metadata.checklist || {};
  1265→
  1266→    return {
  1267→      success: true,
  1268→      data: {
  1269→        rentalId,
  1270→        photos: {
  1271→          passportMainpage: { path: photoPaths.passportMainpage, signedUrl: signedUrls.passportMainpage || null },
  1272→          passportRegistration: { path: photoPaths.passportRegistration, signedUrl: signedUrls.passportRegistration || null },
  1273→          driversLicence: { path: photoPaths.driversLicence, signedUrl: signedUrls.driversLicence || null },
  1274→        },
  1275→        ocrData: {
  1276→          fullName: secrets?.renter_full_name || null,
  1277→          passport: secrets?.renter_passport || null,
  1278→          passportIssuedBy: secrets?.renter_passport_issued_by || null,
  1279→          passportIssueDate: secrets?.renter_passport_issue_date || null,
  1280→          birthDate: secrets?.renter_birth_date || null,
  1281→          registration: secrets?.renter_registration || null,
  1282→          driverLicense: secrets?.renter_driver_license || null,
  1283→        },
  1284→        checklist: {
  1285→          passportVerified: !!checklist.passport_verified,
  1286→          licenseVerified: !!checklist.license_verified,
  1287→          equipmentHandover: !!(checklist.equipment_handover?.keys && checklist.equipment_handover?.helmet),
  1288→          odometerBefore: !!checklist.odometer_before,
  1289→          datesConfirmed: !!checklist.dates_confirmed,
  1290→          paymentVerified: !!checklist.payment_verified,
  1291→        },
  1292→      },
  1293→    };
  1294→  } catch (error) {
  1295→    logger.error("[getRentalDocVerification] failed:", error);
  1296→    return { success: false, error: error instanceof Error ? error.message : "Unknown error" };
  1297→  }
