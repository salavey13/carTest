   126→// Derive flow type per-item from pricing + rawSpecs rent/sale flags
   127→// "rental" → shows 1/3/7 days, packages, auction, rental strip
   128→// "order"  → simple select/purchase flow, no rental UI
   129→function getItemFlowType(item: CatalogItemVM): FlowType {
   130→  const rs = item.rawSpecs;
   131→  const hasExplicitRent = rs != null && "rent" in rs;
   132→  const hasExplicitSale = rs != null && "sale" in rs;
   133→  const rentEnabled = hasExplicitRent ? isSpecEnabled(rs!.rent) : true;
   134→  const saleEnabled = hasExplicitSale ? isSpecEnabled(rs!.sale) : false;
   135→
   136→  // rent explicitly disabled → order flow (even if daily_price column has legacy value)
   137→  if (!rentEnabled) return "order";
   138→  // sale explicitly enabled but rent NOT explicitly set → sale-only
   139→  // (prevents "Забронировать" on bikes where daily_price exists from import
   140→  // but the bike is actually only for sale per operator flag)
   141→  if (saleEnabled && !hasExplicitRent) return "order";
   142→  // Items with rental pricing and rent enabled → rental flow
   143→  return item.pricePerDay > 0 ? "rental" : "order";
   144→}

   174→// ── Spec chip builders ──
   175→// The catalog card shows up to 3 spec chips. The OLD logic just took
   176→// Object.entries(rawSpecs).slice(0,3) — which landed on whatever fields
   177→// happened to be first in the JSON (year, chain type, rental count, …)
   178→// regardless of how informative they were. These builders deliberately
   179→// pick the three most decision-relevant specs, split by propulsion:
   180→//   electric → power (kW), range (km), battery (А·ч)
   181→//   gas/ICE  → power (л.с.), engine (см³), top speed (км/ч)
   182→// Fallbacks fill the remaining slots so we still show 3 chips when a
   183→// primary field is missing (e.g. electric bike with no range → show
   184→// top speed instead). Falls back to the raw specs array only if the
   185→// bike has none of the known keys at all.
   186→
   187→type SpecChipDef = {
   188→  keys: string[];
   189→  icon: string;
   190→  /** format the matched value into chip text */
   191→  format: (raw: string | number) => string;
   192→  /** optional: extract a sub-value from a rich string (e.g. "120Ah" out of "72V 120Ah (Li-ion)") */
   193→  extract?: (raw: string) => string | null;
   194→};
   195→
   196→/** Parse a numeric spec, tolerating Russian units / ranges. Returns the
   197→ *  rounded integer as a string, or — for range-like values like "90-150"
   198→ *  or unparseable strings — the original string unchanged. */
   199→function formatSpecNum(raw: string | number): string {
   200→  if (typeof raw === "number" && Number.isFinite(raw)) return String(Math.round(raw));
   201→  const s = String(raw).trim();
   202→  if (!s) return s;
   203→  // Range-like ("90-150", "70–110", "90 to 150") → keep verbatim
   204→  if (/[-–]\d/.test(s) || /\s(to|до)\s/i.test(s)) return s;
   205→  const cleaned = s.replace(",", ".").replace(/[^\d.]/g, "");
   206→  const n = Number(cleaned);
   207→  return cleaned && Number.isFinite(n) ? String(Math.round(n)) : s;
   208→}
   209→
   210→/** Pull the Ah figure out of a battery string like "72V 120Ah (Li-ion, съёмная)". */
   211→function extractAh(raw: string | number): string | null {
   212→  const s = typeof raw === "number" ? String(raw) : raw;
   213→  const m = s.match(/(\d+(?:[.,]\d+)?)\s*[AaАа]\s*[HhЧч]/);
   214→  if (!m) return null;
   215→  const ah = Number(m[1].replace(",", "."));
   216→  return Number.isFinite(ah) ? `${Math.round(ah)} А·ч` : null;
   217→}
   218→
   219→function pickChip(rawSpecs: Record<string, unknown>, def: SpecChipDef): { icon: string; text: string } | null {
   220→  for (const key of def.keys) {
   221→    if (!(key in rawSpecs)) continue;
   222→    const v = rawSpecs[key];
   223→    if (v === undefined || v === null || v === "") continue;
   224→    if (typeof v === "object") continue; // skip nested (battery_options, etc.)
   225→    const s = typeof v === "string" ? v.trim() : String(v);
   226→    if (!s) continue;
   227→    if (def.extract) {
   228→      const extracted = def.extract(s);
   229→      if (extracted) return { icon: def.icon, text: extracted };
   230→      continue; // extraction failed → try next key
   231→    }
   232→    return { icon: def.icon, text: def.format(s) };
   233→  }
   234→  return null;
   235→}

   237→// Electric: power → range → battery, with top speed / charge time as fillers.
   238→const ELECTRIC_CHIP_DEFS: SpecChipDef[] = [
   239→  { keys: ["power_kw", "motor_nominal_kw", "motor_kw", "motor_peak_kw"], icon: "", format: (v) => `${formatSpecNum(v)} кВт` },
   240→  { keys: ["power_w"], icon: "", format: (v) => `${Math.round((Number(String(v).replace(/[^\d.]/g, "")) || 0) / 1000)} кВт` },
   241→  { keys: ["range_km", "max_range_km"], icon: "", format: (v) => `${formatSpecNum(v)} км` },
   242→  { keys: ["battery", "battery_capacity", "battery_capacity_kwh"], icon: "", format: (v) => String(v), extract: extractAh },
   243→  { keys: ["top_speed_kmh", "max_speed_kmh"], icon: "", format: (v) => `${formatSpecNum(v)} км/ч` },
   244→  { keys: ["charge_time_h", "charging_time_h"], icon: "", format: (v) => `${formatSpecNum(v)} ч` },
   245→];
   246→
   247→// Gas/ICE: power → displacement → top speed, with torque as filler.
   248→const GAS_CHIP_DEFS: SpecChipDef[] = [
   249→  { keys: ["power_hp", "horsepower", "hp", "bike_power_hp"], icon: "", format: (v) => `${formatSpecNum(v)} л.с.` },
   250→  { keys: ["engine_cc", "bike_engine_cc", "displacement_cc", "cc"], icon: "", format: (v) => `${formatSpecNum(v)} см³` },
   251→  { keys: ["top_speed_kmh", "max_speed_kmh"], icon: "", format: (v) => `${formatSpecNum(v)} км/ч` },
   252→  { keys: ["torque_nm"], icon: "", format: (v) => `${formatSpecNum(v)} Н·м` },
   253→];
   254→
   255→function buildChipsFromDefs(rawSpecs: Record<string, unknown>, defs: SpecChipDef[]): Array<{ icon: string; text: string }> {
   256→  const chips: Array<{ icon: string; text: string }> = [];
   257→  for (const def of defs) {
   258→    if (chips.length >= 3) break;
   259→    const chip = pickChip(rawSpecs, def);
   260→    // Skip if extraction failed, OR if an earlier def already produced the
   261→    // same text (e.g. a bike exposes BOTH power_kw=3 and power_w=3000, which
   262→    // both render as "3 кВт" — keep only the first).
   263→    if (chip && !chips.some((c) => c.text === chip.text)) chips.push(chip);
   264→  }
   265→  return chips;
   266→}

   268→function getVisibleSpecChips(item: CatalogItemVM): Array<{ icon: string; text: string }> {
   269→  type ItemWithSpecs = CatalogItemVM & {
   270→    specs?: Array<{ key?: string; label?: string; value?: string; icon?: string }>;
   271→    rawSpecs?: Record<string, unknown>;
   272→    specChips?: Array<{ icon: string; text: string }>;
   273→    subtitle?: string;
   274→    category?: string;
   275→    type?: string;
   276→  };
   277→
   278→  const withSpecs = item as ItemWithSpecs;
   279→
   280→  // 1. Pre-formatted specChips (explicit override from the data pipeline)
   281→  if (Array.isArray(withSpecs.specChips) && withSpecs.specChips.length > 0) {
   282→    return withSpecs.specChips.slice(0, 3);
   283→  }
   284→
   285→  const rawSpecs =
   286→    withSpecs.rawSpecs && typeof withSpecs.rawSpecs === "object"
   287→      ? (withSpecs.rawSpecs as Record<string, unknown>)
   288→      : {};
   289→
   290→  // Equipment spec labels in Russian (for catalog cards)
   291→  const EQUIPMENT_SPEC_LABELS: Record<string, string> = {
   292→    size: "Размер",
   293→    sizes: "Размеры",
   294→    material: "Материал",
   295→    materials: "Материалы",
   296→    badge: "Особенность",
   297→    features: "Особенности",
   298→    safety: "Сертификация",
   299→    category: "Категория",
   300→    color: "Цвет",
   301→    colors: "Цвета",
   302→    weight: "Вес",
   303→    brand: "Бренд",
   304→  };
   305→
   306→  // Get Russian label from spec_labels in rawSpecs
   307→  const getRussianLabel = (key: string): string => {
   308→    const specLabels = (rawSpecs.spec_labels as Record<string, string> | undefined);
   309→    return specLabels?.[key] || EQUIPMENT_SPEC_LABELS[key] || key;
   310→  };
   311→
   312→  // 2. Equipment items: show size as highlighted spec
   313→  if (withSpecs.type === "equipment" || rawSpecs.category) {
   314→    const chips: Array<{ icon: string; text: string }> = [];
   315→
   316→    // Size is the most important spec for equipment
   317→    if (Array.isArray(rawSpecs.sizes) && rawSpecs.sizes.length > 0) {
   318→      const sizes = rawSpecs.sizes as string[];
   319→      chips.push({ icon: "", text: sizes.slice(0, 3).join(" / ") });
   320→    }
   321→
   322→    // Show badge if available (use Russian label)
   323→    if (typeof rawSpecs.badge === "string" && rawSpecs.badge) {
   324→      const badgeLabel = getRussianLabel("badge");
   325→      chips.push({ icon: "", text: `${badgeLabel}: ${rawSpecs.badge as string}` });
   326→    }
   327→
   328→    // Show material if available (use Russian label)
   329→    if (typeof rawSpecs.materials === "string" && rawSpecs.materials) {
   330→      const materialLabel = getRussianLabel("materials");
   331→      chips.push({ icon: "", text: rawSpecs.materials as string });
   332→    }
   333→
   334→    // Fallback to features array if we don't have 3 chips yet
   335→    if (chips.length < 3 && Array.isArray(rawSpecs.features) && rawSpecs.features.length > 0) {
   336→      const features = rawSpecs.features as string[];
   337→      chips.push(...features.slice(0, 3 - chips.length).map(f => ({ icon: "", text: f })));
   338→    }
   339→
   340→    return chips.slice(0, 3);
   341→  }
   342→
   343→  // 3. Propulsion-aware deliberate picks for bikes (power / range-or-cc / battery-or-speed)
   344→  if (Object.keys(rawSpecs).length > 0) {
   345→    const propulsion = getCatalogPropulsionSegment({
   346→      title: item.title,
   347→      subtitle: withSpecs.subtitle,
   348→      description: item.description,
   349→      category: withSpecs.category,
   350→      rawSpecs,
   351→    });
   352→    const defs = propulsion === "gas" ? GAS_CHIP_DEFS : ELECTRIC_CHIP_DEFS;
   353→    const chips = buildChipsFromDefs(rawSpecs, defs);
   354→    if (chips.length > 0) return chips;
   355→  }
   356→
   357→  // 4. Last-resort fallback: structured specs array (label/value pairs)
   358→  if (Array.isArray(withSpecs.specs) && withSpecs.specs.length > 0) {
   359→    return withSpecs.specs
   360→      .filter((s) => s.value || s.label)
   361→      .slice(0, 3)
   362→      .map((s) => ({
   363→        icon: ("icon" in s && typeof s.icon === "string") ? s.icon : specIconForKey(s.label ?? ""),
   364→        text: s.value ?? s.label ?? "",
   365→      }));
   366→  }
   367→
   368→  return [];
   369→}
