    32→const normalizeVin = (value: unknown) => (typeof value === "string" ? value.trim().toUpperCase() : "");
    33→
    34→const VIN_ALLOWED = "ABCDEFGHJKLMNPRSTUVWXYZ0123456789";
    35→
    36→const buildSyntheticVin = (vehicle: Vehicle) => {
    37→  const raw = `${vehicle.make ?? ""}${vehicle.model ?? ""}${vehicle.year ?? ""}${vehicle.id ?? ""}`.toUpperCase();
    38→  const mapped = raw
    39→    .replace(/I/g, "1")
    40→    .replace(/O/g, "0")
    41→    .replace(/Q/g, "0")
    42→    .replace(/[^A-Z0-9]/g, "");
    43→  const normalized = mapped
    44→    .split("")
    45→    .map((char) => (VIN_ALLOWED.includes(char) ? char : "X"))
    46→    .join("");
    47→  return (normalized + "CARTESTVIN00000000").slice(0, 17);
    48→};

   242→  const { dbUser, userCrewInfo, isAdmin, isLoading: appContextLoading } = useAppContext();

   246→
   247→  const [isTrulyAdmin, setIsTrulyAdmin] = useState<boolean>(false);
   248→  const [userVehicles, setUserVehicles] = useState<Vehicle[]>([]);
   249→  const [selectedVehicle, setSelectedVehicle] = useState<Vehicle | null>(null);
   250→  const [isFetchingVehicles, setIsFetchingVehicles] = useState(false);
   251→  const [typeFilter, setTypeFilter] = useState<"all" | "bike" | "car">("all");
   252→  const [searchQuery, setSearchQuery] = useState("");
   253→  const [showShortcuts, setShowShortcuts] = useState(false);
   254→  const [isBulkFillingVin, setIsBulkFillingVin] = useState(false);
   255→
   256→  const fetchUserVehicles = useCallback(async (userId: string) => {
   257→    setIsFetchingVehicles(true);
   258→    try {
   259→      const { data, error, success } = await getEditableVehiclesForUser(userId);
   260→      
   261→      if (!success || error) {
   262→        throw new Error(error || "Не удалось получить список техники.");
   263→      }
   264→      
   265→      setUserVehicles((data || []).filter(v => v.type === 'bike' || v.type === 'car'));
   266→      
   267→      if(editId) {
   268→        const vehicleToEdit = data?.find(v => v.id === editId);
   269→        if (vehicleToEdit) {
   270→            setSelectedVehicle(vehicleToEdit);
   271→            toast.info(`Загружен для редактирования: ${vehicleToEdit.make} ${vehicleToEdit.model}`);
   272→        } else {
   273→            toast.error(`Транспорт с ID ${editId} не найден в вашем гараже или команде.`);
   274→        }
   275→      }
   276→    } catch (error) {
   277→      toast.error(error instanceof Error ? error.message : "Не удалось загрузить вашу технику.");
   278→    } finally {
   279→      setIsFetchingVehicles(false);
   280→    }
   281→  }, [editId]);
   282→
   283→  useEffect(() => {
   284→    if (!appContextLoading && dbUser?.user_id) {
   285→      if (typeof isAdmin === 'function') {
   286→        setIsTrulyAdmin(isAdmin());
   287→      }
   288→      fetchUserVehicles(dbUser.user_id);
   289→    }
   290→  }, [appContextLoading, dbUser, isAdmin, fetchUserVehicles]);
   291→
   292→  // Keyboard shortcuts
   293→  useEffect(() => {
   294→    const handler = (e: KeyboardEvent) => {
   295→      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;
   296→      
   297→      if (e.key === '?' && e.shiftKey) {
   298→        e.preventDefault();
   299→        setShowShortcuts(prev => !prev);
   300→      }
   301→      if (e.key === 'n' || e.key === 'N') {
   302→        e.preventDefault();
   303→        setSelectedVehicle(null);
   304→        router.push('/admin');
   305→      }
   306→      if (e.key === 'f' || e.key === 'F') {
   307→        e.preventDefault();
   308→        const input = document.querySelector('input[placeholder="Поиск в гараже..."]') as HTMLInputElement;
   309→        input?.focus();
   310→      }
   311→    };
   312→    
   313→    window.addEventListener('keydown', handler);
   314→    return () => window.removeEventListener('keydown', handler);
   315→  }, [router]);
   316→
   317→  const visibleVehicles = useMemo(() => {
   318→    return userVehicles
   319→      .filter(v => typeFilter === "all" ? true : v.type === typeFilter)
   320→      .filter(v => {
   321→        if (!searchQuery.trim()) return true;
   322→        const q = searchQuery.toLowerCase();
   323→        return (
   324→          v.make?.toLowerCase().includes(q) ||
   325→          v.model?.toLowerCase().includes(q) ||
   326→          v.year?.toString().includes(q) ||
   327→          v.specs?.vin?.toLowerCase().includes(q)
   328→        );
   329→      });
   330→  }, [userVehicles, typeFilter, searchQuery]);
   331→
   332→
   333→
   334→  const vinAudit = useMemo(() => {
   335→    const target = userVehicles.filter(v => v.type === "bike" || v.type === "car");
   336→    const missing = target.filter(v => !normalizeVin(v.specs?.vin));
   337→    return {
   338→      total: target.length,
   339→      withVin: target.length - missing.length,
   340→      missing,
   341→    };
   342→  }, [userVehicles]);
   343→
   344→  const handleBulkFillVin = useCallback(async () => {
   345→    if (!dbUser?.user_id || !vinAudit.missing.length) return;
   346→    setIsBulkFillingVin(true);
   347→    try {
   348→      for (const vehicle of vinAudit.missing) {
   349→        const syntheticVin = buildSyntheticVin(vehicle);
   350→        const res = await fetch('/api/cars', {
   351→          method: 'PUT',
   352→          headers: { 'Content-Type': 'application/json' },
   353→          body: JSON.stringify({
   354→            ...vehicle,
   355→            specs: {
   356→              ...(vehicle.specs || {}),
   357→              vin: syntheticVin,
   358→            },
   359→          }),
   360→        });
   361→        if (!res.ok) {
   362→          throw new Error(`VIN bulk-fill failed for ${vehicle.make} ${vehicle.model}`);
   363→        }
   364→      }
   365→      toast.success(`VIN заполнен для ${vinAudit.missing.length} карточек`);
   366→      await fetchUserVehicles(dbUser.user_id);
   367→    } catch (error) {
   368→      toast.error(error instanceof Error ? error.message : 'Не удалось выполнить bulk-fill VIN');
   369→    } finally {
   370→      setIsBulkFillingVin(false);
   371→    }
   372→  }, [dbUser?.user_id, vinAudit.missing, fetchUserVehicles]);

   374→    setSelectedVehicle(vehicle);
   375→    if (vehicle) {
   376→      router.push(`/admin?edit=${vehicle.id}`, { scroll: false });
   377→      toast.success(`Редактирование: ${vehicle.make} ${vehicle.model}`);
   378→    }
   379→  };
   380→  
   381→  const handleFormSuccess = () => {
   382→    if (dbUser?.user_id) {
   383→      fetchUserVehicles(dbUser.user_id);
   384→    }
   385→    if (!editId) {
   386→      setSelectedVehicle(null);
   387→    }
   388→    toast.success("Изменения сохранены");
   389→  };
   390→
   391→  if (appContextLoading) { 
   392→    return <Loading text="ПРОВЕРКА ДОСТУПА..." />;
   393→  }
   394→  
   395→  const crewSlug = userCrewInfo?.slug || "vip-bike";
   396→
   397→  return (

   400→      <div className="fixed inset-0 z-[-1] opacity-20">
   401→        <Image

   403→          alt="Admin Background"
   404→          fill
   405→          className="object-cover animate-pan-zoom"

   409→        {/* Grid overlay for cyberpunk feel */}
   410→        <div className="absolute inset-0 bg-[linear-gradient(rgba(124,244,120,0.03)_1px,transparent_1px),linear-gradient(90deg,rgba(124,244,120,0.03)_1px,transparent_1px)] bg-[size:40px_40px]"></div>
   411→      </div>
   412→
   413→      <main className="container mx-auto pt-4 sm:pt-6 px-3 sm:px-4 relative z-10 pb-16 max-w-7xl">
   414→        {/* Header */}
   415→        <motion.div
   416→          initial={{ opacity: 0, y: -10 }}
   417→          animate={{ opacity: 1, y: 0 }}
   418→          className="mb-4 sm:mb-6"
   419→        >
   420→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
