   151→export const createOrUpdateUser = async (userId: string, userInfo: Partial<WebAppUser & { role?: DbUser['role']; status?: DbUser['status']; metadata?: Record<string, any> }>): Promise<DbUser | null> => {
   152→     if (!userId) {

   155→    }
   156→     if (!supabaseAdmin) {
   157→         logger.error("[SupabaseHook] createOrUpdateUser: Admin client unavailable.");
   158→         return null;
   159→     }
   160→
   161→    logger.info("[SupabaseHook] Attempting to create or update user:", { userId, username: userInfo.username });
   162→
   163→    try {
   164→        const userData: Partial<DbUser> = { 
   165→            user_id: userId, 
   166→            username: userInfo.username || null,
   167→            full_name: `${userInfo.first_name || ""} ${userInfo.last_name || ""}`.trim() || null,
   168→            avatar_url: userInfo.photo_url || null,
   169→            language_code: userInfo.language_code || null,
   170→            ...(userInfo.role && { role: userInfo.role }),
   171→            ...(userInfo.status && { status: userInfo.status }),
   172→            ...(userInfo.metadata !== undefined && { metadata: userInfo.metadata }), 
   173→            updated_at: new Date().toISOString(),
   174→        };
   175→        
   176→        Object.keys(userData).forEach(key => {
   177→            const k = key as keyof typeof userData;
   178→            if (userData[k] === undefined && k !== 'metadata') {
   179→                 delete userData[k];
   180→            }
   181→        });
   182→        
   183→        const { data, error } = await supabaseAdmin
   184→            .from("users") 
   185→            .upsert(userData as Database["public"]["Tables"]["users"]["Insert"], { 
   186→                onConflict: 'user_id', 
   187→            })
   188→            .select("*, metadata") 
   189→            .single();
   190→
   191→        if (error) {
   192→            logger.error(`[SupabaseHook] Error upserting user ${userId}:`, error);
   193→            throw error;
   194→        }
   195→
   196→        if (!data) {
   197→             logger.error(`[SupabaseHook] Upsert for user ${userId} returned no data.`);
   198→            throw new Error("Failed to get user data after upsert.");
   199→        }
   200→
   201→        logger.info(`[SupabaseHook] User ${userId} upserted successfully.`);
   202→        return data;
   203→    } catch (catchError) {
   204→        logger.error(`[SupabaseHook] Exception in createOrUpdateUser for ${userId}:`, catchError);
   205→        return null;
   206→    }

   240→    }
   241→

   250→const VECTOR_DIMENSIONS = 384; 
   251→
   252→function generateSimplifiedFallbackEmbedding(text: string): number[] {
   253→   logger.warn("Generating simplified FALLBACK embedding for text:", text.substring(0, 50) + "...");
   254→   const embedding = new Array(VECTOR_DIMENSIONS).fill(0);
   255→   if (!text) return embedding;
   256→   const words = text.toLowerCase().split(/\W+/).filter(Boolean);
   257→   words.forEach((word, i) => {
   258→       let hash = 0;
   259→       for (let j = 0; j < word.length; j++) {
   260→           hash = (hash * 31 + word.charCodeAt(j)) | 0;
   261→       }
   262→       embedding[Math.abs(hash + i) % VECTOR_DIMENSIONS] += 1;
   263→   });
   264→    const magnitude = Math.sqrt(embedding.reduce((sum, val) => sum + val * val, 0));
   265→    return magnitude > 0 ? embedding.map(val => val / magnitude) : embedding;
   266→}

   268→export async function generateCarEmbedding(
   269→  mode: 'single' | 'batch' | 'create' = 'batch',
   270→  payload?: {
   271→    carId?: string; 
   272→    carData?: {
   273→        make: string; model: string; description: string; specs: Record<string, any>; 
   274→        owner_id?: string; daily_price?: number; image_url?: string; rent_link?: string; status?: string;
   275→    };
   276→  }
   277→): Promise<{ success: boolean; message: string; carId?: string; error?: string }> {
   278→    const serviceRoleKey = getServiceRoleKey();
   279→    if (!serviceRoleKey || !supabaseUrl) {
   280→        return { success: false, message: "Service role key or Supabase URL missing.", error:"Configuration error." };
   281→    }
   282→    if (!supabaseAdmin) {
   283→        return { success: false, message: "Admin client is not available.", error: "Configuration error." };
   284→    }
   285→
   286→    let endpoint: string;
   287→    let requestBody: any = {};
   288→    const headers = {
   289→        Authorization: `Bearer ${serviceRoleKey}`,
   290→        "Content-Type": "application/json",
   291→    };
   292→
   293→    try {
   294→        switch (mode) {
   295→            case 'single':
   296→                if (!payload?.carId) throw new Error("carId is required for single mode");
   297→                endpoint = `${supabaseUrl}/functions/v1/generate-embeddings`;
   298→                requestBody = { mode: 'single', carId: payload.carId };
   299→                break;
   300→            case 'create':
   301→                if (!payload?.carData) throw new Error("carData is required for create mode");
   302→                endpoint = `${supabaseUrl}/functions/v1/generate-embeddings`;
   303→                requestBody = {
   304→                    mode: 'create',
   305→                    make: payload.carData.make,
   306→                    model: payload.carData.model,
   307→                    description: payload.carData.description,
   308→                    specs: payload.carData.specs,
   309→                 };
   310→                break;
   311→            case 'batch':
   312→            default:
   313→                endpoint = `${supabaseUrl}/functions/v1/generate-embeddings`;
   314→                requestBody = { mode: 'batch' };
   315→                break;
   316→        }
   317→
   318→        logger.info(`Calling Edge Function 'generate-embeddings' (mode: ${mode})...`);
   319→        const response = await fetch(endpoint, {
   320→            method: "POST",
   321→            headers,
   322→            body: JSON.stringify(requestBody),
   323→        });
   324→
   325→        const responseText = await response.text();
   326→        if (!response.ok) {
   327→            logger.error(`Edge Function 'generate-embeddings' (${mode}) failed (${response.status}): ${responseText}`);
   328→            throw new Error(`Edge Function failed (${response.status}): ${responseText}`);
   329→        }
   330→
   331→        const result = JSON.parse(responseText);
   332→        logger.info(`Edge Function 'generate-embeddings' (${mode}) success: ${result.message}`);
   333→
   334→        let createdCarId = result.carId; 
   335→        if (mode === 'create' && createdCarId && payload?.carData) {
   336→            const { owner_id, daily_price, image_url, rent_link, status } = payload.carData;
   337→            const fieldsToUpdate: Partial<DbCar> = {};
   338→            if (owner_id !== undefined) fieldsToUpdate.owner_id = owner_id;
   339→            if (daily_price !== undefined) fieldsToUpdate.daily_price = daily_price;
   340→            if (image_url !== undefined) fieldsToUpdate.image_url = image_url;
   341→            if (rent_link !== undefined) fieldsToUpdate.rent_link = rent_link;
   342→            if (status !== undefined) fieldsToUpdate.status = status;
   343→
   344→            if (Object.keys(fieldsToUpdate).length > 0) {
   345→                logger.info(`Updating additional fields for new car ${createdCarId}...`, fieldsToUpdate);
   346→                const { error: updateError } = await supabaseAdmin
   347→                    .from("cars")
   348→                    .update(fieldsToUpdate)
   349→                    .eq("id", createdCarId);
   350→
   351→                 if (updateError) {
   352→                     logger.error(`Failed to update additional fields for car ${createdCarId}: ${updateError.message}`);
   353→                     result.message += ` (Warning: Failed to update some additional fields: ${updateError.message})`;
   354→                 } else {
   355→                     logger.info(`Successfully updated additional fields for car ${createdCarId}.`);
   356→                 }
   357→            }
   358→        }
   359→        return { success: true, message: result.message, carId: createdCarId };
   360→
   361→    } catch (error) {
   362→        logger.error(`generateCarEmbedding (${mode}) failed, attempting fallback:`, error);
   363→        try {
   364→            let combinedText: string | null = null;
   365→            let carIdToUpdate: string | undefined = undefined;
   366→
   367→            if (mode === 'single' && payload?.carId) {
   368→                const { data: car, error: fetchError } = await supabaseAdmin.from("cars").select("make, model, description, specs").eq("id", payload.carId).single();
   369→                if (fetchError || !car) { throw new Error(`Fallback failed: Cannot fetch car ${payload.carId}. ${fetchError?.message}`); }
   370→                combinedText = `${car.make} ${car.model} ${car.description || ''} ${JSON.stringify(car.specs || {})}`;
   371→                carIdToUpdate = payload.carId;
   372→            } else if (mode === 'create' && payload?.carData) {
   373→                 combinedText = `${payload.carData.make} ${payload.carData.model} ${payload.carData.description || ''} ${JSON.stringify(payload.carData.specs || {})}`;
   374→            } else {
   375→                 return { success: false, message: "Edge function failed, fallback not supported for batch mode.", error: error instanceof Error ? error.message : String(error) };
   376→            }
   377→
   378→            if (!combinedText) {
   379→                throw new Error("Fallback failed: Could not generate combined text for embedding.");
   380→            }
   381→            const embedding = generateSimplifiedFallbackEmbedding(combinedText);
   382→
   383→            if (carIdToUpdate) {
   384→                 const { error: updateError } = await supabaseAdmin.from("cars").update({ embedding }).eq("id", carIdToUpdate);
   385→                 if (updateError) throw new Error(`Fallback failed: Cannot update embedding for ${carIdToUpdate}. ${updateError.message}`);
   386→                 logger.info(`Successfully updated car ${carIdToUpdate} with fallback embedding.`);
   387→                 return { success: true, message: "Edge function failed, fallback embedding generated and updated.", carId: carIdToUpdate };
   388→            } else if (mode === 'create' && payload?.carData) {
   389→                 const carInsertData: Database["public"]["Tables"]["cars"]["Insert"] = {
   390→                        make: payload.carData.make, model: payload.carData.model, description: payload.carData.description,
   391→                        specs: payload.carData.specs, owner_id: payload.carData.owner_id, daily_price: payload.carData.daily_price,
   392→                        image_url: payload.carData.image_url, rent_link: payload.carData.rent_link,
   393→                        embedding, is_test_result: false, 
   394→                        status: payload.carData.status || 'available', 
   395→                 };
   396→                 const { data: newCar, error: insertError } = await supabaseAdmin
   397→                    .from("cars")
   398→                    .insert(carInsertData)
   399→                    .select("id")
   400→                    .single();
   401→                 if (insertError || !newCar) throw new Error(`Fallback failed: Cannot create car. ${insertError?.message}`);
   402→                 logger.info(`Successfully created car ${newCar.id} with fallback embedding.`);
   403→                 return { success: true, message: "Edge function failed, fallback embedding generated and car created.", carId: newCar.id };
   404→            }
   405→             return { success: false, message: "Edge function failed, and fallback logic path unclear.", error: error instanceof Error ? error.message : String(error) };
   406→        } catch (fallbackError) {
   407→            logger.error("Fallback embedding generation also failed:", fallbackError);
   408→            return { success: false, message: "Edge function and fallback embedding failed.", error: fallbackError instanceof Error ? fallbackError.message : String(fallbackError) };
   409→        }
   410→    }
   411→}

   424→export const searchCars = async (embedding: number[], limit: number = 5): Promise<{ success: boolean; data?: CarResult[]; error?: string }> => { 
   425→    if (!embedding || embedding.length !== VECTOR_DIMENSIONS) {
   426→        return { success: false, error: `Invalid embedding provided (length ${embedding?.length}, expected ${VECTOR_DIMENSIONS})` };
   427→    }
   428→    if (!supabaseAnon) {
   429→        return { success: false, error: "Anon client is not available for search." };
   430→    }
   431→    try {
   432→        const { data, error } = await supabaseAnon.rpc("search_cars", {
   433→            query_embedding: embedding,
   434→            match_count: limit,
   435→        });
   436→
   437→        if (error) {
   438→            logger.error("Error calling search_cars RPC:", error);
   439→            throw error;
   440→        }
   441→        const formattedData: CarResult[] = (data || []).map((item: any) => ({
   442→            id: item.id,
   443→            make: item.make,
   444→            model: item.model,
   445→            similarity: item.similarity,
   446→            description: item.description,
   447→            image_url: item.image_url,
   448→            daily_price: item.daily_price,
   449→            rent_link: item.rent_link || `/rent/${item.id}`,
   450→        }));
   451→        return { success: true, data: formattedData };
   452→    } catch (error) {
   453→        logger.error("Error searching cars:", error);
   454→        return { success: false, error: error instanceof Error ? error.message : "Failed to search cars" };
   455→    }

   594→export const createInvoice = async (
   595→    type: string,
   596→    id: string, 
   597→    userId: string,
   598→    amount: number,
   599→    subscriptionId?: number, 
   600→    metadata: Record<string, any> = {}
   601→): Promise<{ success: boolean; data?: DbInvoice; error?: string }> => {
   602→    if (!supabaseAdmin) return { success: false, error: "Admin client not available."};
   603→    if (!id || !userId || amount == null || !type) return { success: false, error: "Missing required parameters for invoice creation." };
   604→
   605→    try {
   606→        const { data, error } = await supabaseAdmin
   607→            .from("invoices")
   608→            .insert({
   609→                id: id,
   610→                user_id: userId,
   611→                type: type,
   612→                amount: amount, 
   613→                status: 'pending', 
   614→                subscription_id: subscriptionId, 
   615→                metadata: metadata,
   616→                created_at: new Date().toISOString(),
   617→                updated_at: new Date().toISOString()
   618→             })
   619→            .select() 
   620→            .single(); 
   621→
   622→        if (error) {
   623→            if (error.code === '23505') { 
   624→                 logger.warn(`Invoice with ID ${id} already exists.`);
   625→                 const existing = await getInvoiceById(id);
   626→                 if (existing.success && existing.data) return { success: true, data: existing.data };
   627→                 return { success: false, error: `Invoice ID ${id} already exists, but failed to retrieve it.` };
   628→            }
   629→            logger.error(`Error inserting invoice ${id}:`, {
   630→              message: error.message,
   631→              details: error.details,
   632→              hint: error.hint,
   633→              code: error.code,
   634→            });
   635→            throw error; 
   636→        }
   637→        if (!data) return { success: false, error: "Invoice creation returned no data." };
   638→
   639→        logger.info(`Invoice ${id} created successfully for user ${userId}.`);
   640→        return { success: true, data };
   641→    } catch (error) {
   642→        const castError = error as any;
   643→        logger.error(`Exception creating invoice ${id} for user ${userId}:`, {
   644→            message: castError.message,
   645→            details: castError.details,
   646→            hint: castError.hint,
   647→            code: castError.code,
   648→            fullError: castError,
   649→          });
   650→        return { success: false, error: castError instanceof Error ? castError.message : "Failed to create invoice" };
   651→    }

   851→export const uploadImage = async (bucketName: string, file: File, fileName?: string, options?: { upsert?: boolean }): Promise<{ success: boolean; publicUrl?: string; error?: string }> => {
   852→     if (!supabaseAdmin) return { success: false, error: "Admin client not available."};
   853→     if (!bucketName || !file) return { success: false, error: "Bucket name and file are required." };
   854→
   855→    try {
   856→        const fileExt = file.name.split(".").pop();
   857→        if (!fileExt) {
   858→            return { success: false, error: "Invalid file name (no extension)." };
   859→        }
   860→        const finalFileName = fileName || `${crypto.randomUUID()}.${fileExt}`;
   861→        const filePath = `${finalFileName}`; 
   862→
   863→        logger.info(`Uploading file '${file.name}' to bucket '${bucketName}' as '${filePath}'...`);
   864→
   865→        const { error: uploadError } = await supabaseAdmin.storage
   866→            .from(bucketName)
   867→            .upload(filePath, file, {
   868→                cacheControl: '3600', 
   869→                // allow callers to override (e.g. editor re-saving to carpix/<id>/image_1.jpg)
   870→                upsert: options?.upsert ?? false
   871→            });
   872→
   873→        if (uploadError) {
   874→            logger.error(`Error uploading image to ${bucketName}/${filePath}:`, uploadError);
   875→            throw uploadError;
   876→        }
   877→
   878→        const { data: urlData } = supabaseAdmin.storage
   879→            .from(bucketName)
   880→            .getPublicUrl(filePath);
   881→
   882→        if (!urlData?.publicUrl) {
   883→             logger.error(`Failed to get public URL for ${bucketName}/${filePath} after upload.`);
   884→            throw new Error("Failed to get public URL after upload");
   885→        }
   886→
   887→         logger.info(`Image uploaded successfully: ${urlData.publicUrl}`);
   888→        return { success: true, publicUrl: urlData.publicUrl };
   889→
   890→    } catch (error) {
   891→        logger.error(`Error uploading image to bucket ${bucketName}:`, error);
   892→        return { success: false, error: error instanceof Error ? error.message : "Failed to upload image" };
   893→    }
